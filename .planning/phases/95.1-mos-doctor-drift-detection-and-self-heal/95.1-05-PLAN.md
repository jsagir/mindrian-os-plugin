const SKIP_DIRS = new Set([
  '.git', '.mindrian', '.context', '.lazygraph', '.rooms',
  'node_modules', '.next', 'dist', 'build', '.cache',

function findRoomRoot(filePath) {
  // Pure Node port of scripts/post-write detect_room_section()
  // + symlink-safe walk per scripts/hooks/pre-commit-room-minto-guard.sh
  let dir = path.dirname(filePath);
  const visited = new Set();
  let hops = 0;
  while (dir !== '/' && dir !== '.' && hops < 12) {
    let real;
    try { real = fs.realpathSync(dir); } catch (_) { return null; }
    if (visited.has(real)) break;
    visited.add(real);
    if (fs.existsSync(path.join(real, '.room-root'))) return real;
    dir = path.dirname(real);
    hops += 1;
  }
  return null;
}

function readRegistry() {
  const home = process.env.MINDRIAN_ROOMS_HOME || path.join(HOME, 'MindrianRooms');
  const registryPath = path.join(home, '.rooms', 'registry.json');
  if (!fs.existsSync(registryPath)) return null;
  try {
    const raw = fs.readFileSync(registryPath, 'utf8');
    const reg = JSON.parse(raw);
    return { roomsHome: home, registry: reg, registryPath };
  } catch (_) { return null; }
}

function listSubdirs(rootDir, { recursive = true, maxDepth = 8 } = {}) {
  // Returns absolute paths of all directories under rootDir, skipping SKIP_DIRS
  const results = [];
  function walk(dir, depth) {
    if (depth > maxDepth) return;
    let entries;
    try { entries = fs.readdirSync(dir, { withFileTypes: true }); }
    catch (_) { return; }
    for (const entry of entries) {
      if (!entry.isDirectory()) continue;
      if (SKIP_DIRS.has(entry.name)) continue;
      const sub = path.join(dir, entry.name);
      results.push(sub);
      if (recursive) walk(sub, depth + 1);
    }
  }
  walk(rootDir, 0);
  return results;
}

function invokeGenerator(targetDir, { recursive = true, force = false } = {}) {
  const { spawnSync } = require('child_process');
  const generatorPath = path.join(__dirname, 'generate-section-intelligence.cjs');
  if (!fs.existsSync(generatorPath)) {
    return { status: 'error', detail: 'generator not found at ' + generatorPath };
  }
  const args = [generatorPath, targetDir];
  if (recursive) args.push('--recursive');
  if (force) args.push('--force');
  const res = spawnSync('node', args, { encoding: 'utf8', timeout: 30000 });
  return {
    status: res.status === 0 ? 'ok' : 'error',
    stdout: res.stdout || '',
    stderr: res.stderr || '',
    exitCode: res.status,
  };
}

// ── Class B + C: cascade-rooms check ─────────────────────────────────

function checkCascadeRooms() {
  const reg = readRegistry();
  if (!reg) {
    return { status: 'skip', detail: 'no registry at ~/MindrianRooms/.rooms/registry.json (or MINDRIAN_ROOMS_HOME)', classB: null, classC: null };
  }
  const roomsHome = reg.roomsHome;
  const rooms = (reg.registry && reg.registry.rooms) || {};
  const activeName = reg.registry && reg.registry.active;

  // ── Class B: scan every registered room for .room-root sentinel ───
  const missingSentinels = [];
  const okRooms = [];
  for (const [name, info] of Object.entries(rooms)) {
    const relPath = info && info.path;
    if (!relPath) continue;
    const roomPath = path.isAbsolute(relPath) ? relPath : path.join(roomsHome, relPath);
    const sentinelPath = path.join(roomPath, '.room-root');
    if (fs.existsSync(sentinelPath)) {
      okRooms.push({ name, roomPath });
    } else {
      missingSentinels.push({ name, roomPath, sentinelPath });
    }
  }

  // ── Class C: active-room vs PWD-room mismatch ────────────────────
  let classC = { status: 'ok', activeRoom: activeName || null, pwdRoom: null };
  try {
    const pwdRoom = findRoomRoot(path.join(process.cwd(), 'X'));
    // findRoomRoot expects a file path; pass process.cwd() + 'X' so dirname returns cwd
    if (pwdRoom) {
      const pwdRoomName = path.basename(pwdRoom);
      classC.pwdRoom = pwdRoomName;
      // Resolve activeName to its absolute path for compare
      const activeInfo = rooms[activeName];
      if (activeInfo) {
        const activePath = path.isAbsolute(activeInfo.path)
          ? activeInfo.path
          : path.join(roomsHome, activeInfo.path);
        if (path.resolve(pwdRoom) !== path.resolve(activePath)) {
          classC = {
            status: 'warn',
            activeRoom: activeName,
            pwdRoom: pwdRoomName,
            activeMismatch: true,
            detail: `writes to ${pwdRoomName} would be silenced (active=${activeName})`,
          };
        }
      }
    }
  } catch (err) {
    classC = { status: 'error', detail: err.message };
  }

  // ── Class B aggregated status ────────────────────────────────────
  const classB = missingSentinels.length === 0
    ? { status: 'ok', missingSentinels: [], okCount: okRooms.length }
    : { status: 'warn', missingSentinels, okCount: okRooms.length, detail: `${missingSentinels.length} room(s) missing .room-root sentinel` };

  // Combined status: worst of B and C
  let status = 'ok';
  if (classB.status === 'warn' || classC.status === 'warn') status = 'warn';
  if (classB.status === 'error' || classC.status === 'error') status = 'error';

  return { status, classB, classC };
}

// ── Class E: room-md cascade check ───────────────────────────────────

function checkRoomMd() {
  const reg = readRegistry();
  if (!reg) return { status: 'skip', detail: 'no registry; class E scoped to active room', missing: [] };

  const activeName = reg.registry && reg.registry.active;
  if (!activeName) return { status: 'skip', detail: 'no active room', missing: [] };

  const activeInfo = (reg.registry.rooms || {})[activeName];
  if (!activeInfo || !activeInfo.path) return { status: 'skip', detail: 'active room not in registry', missing: [] };

  const roomPath = path.isAbsolute(activeInfo.path)
    ? activeInfo.path
    : path.join(reg.roomsHome, activeInfo.path);

  // Sentinel must exist for class E to apply (Phase 87-01a guard scope)
  if (!fs.existsSync(path.join(roomPath, '.room-root'))) {
    return { status: 'skip', detail: 'active room missing .room-root sentinel (class B finding)', missing: [], roomPath };
  }

  // Walk all subdirectories under the active room (recursive, skip SKIP_DIRS)
  const subdirs = listSubdirs(roomPath, { recursive: true, maxDepth: 8 });
  const missing = [];
  for (const subdir of subdirs) {
    const hasRoom = fs.existsSync(path.join(subdir, 'ROOM.md'));
    const hasMinto = fs.existsSync(path.join(subdir, 'MINTO.md'));
    if (!hasRoom || !hasMinto) {
      const filesMissing = [];
      if (!hasRoom) filesMissing.push('ROOM.md');
      if (!hasMinto) filesMissing.push('MINTO.md');
      missing.push({ section: path.relative(roomPath, subdir), absPath: subdir, files: filesMissing });
    }
  }

  return missing.length === 0
    ? { status: 'ok', missing: [], roomPath, subdirs: subdirs.length }
    : { status: 'warn', missing, roomPath, subdirs: subdirs.length, detail: `${missing.length} dir(s) missing ROOM.md or MINTO.md` };
}

function performRoomMdRecovery(checkResult) {
  if (!checkResult || checkResult.status !== 'warn') {
    return { status: 'skip', detail: 'no class E drift to recover' };
  }
  if (!checkResult.roomPath) {
    return { status: 'error', detail: 'roomPath not set on check result' };
  }
  // Single recursive invocation against the room root resolves all missing dirs
  const result = invokeGenerator(checkResult.roomPath, { recursive: true, force: false });
  if (result.status !== 'ok') {
    return { status: 'error', detail: 'generator failed: ' + (result.stderr || result.stdout || 'no output').slice(0, 200) };
  }
  // Re-run check to confirm
  const afterCheck = checkRoomMd();
  if (afterCheck.status === 'ok') {
    return { status: 'ok', detail: 'all subdirs now have ROOM.md + MINTO.md', generatorOutput: result.stdout };
  }
  return {
    status: 'partial',
    detail: `${afterCheck.missing.length} dir(s) still missing after generation`,
    remaining: afterCheck.missing,
  };
}

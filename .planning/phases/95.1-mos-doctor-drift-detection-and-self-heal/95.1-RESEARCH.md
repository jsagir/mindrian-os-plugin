const POST_WRITE = path.join(REPO, 'scripts', 'post-write');

function runBashHook(scriptPath, envelope, opts) {
  const o = opts || {};
  const res = spawnSync('bash', [scriptPath], {
    encoding: 'utf8',
    input: JSON.stringify(envelope),
    timeout: o.timeoutMs || 5000,
    env: Object.assign({}, process.env, o.env || {}),

### Anti-Patterns to Avoid

- **`╭─ ╮ ╰─ ╯` box chars in any renderer.** SKILL.md §1: "alignment IS the punctuation." Strip from `renderHumanReport` per D-11.
- **`✗` cross mark.** Not in 12-glyph vocabulary. Use `⚠` or `⬜` per SKILL.md §3.
- **Emoji anywhere except `scripts/context-monitor`** (the documented carve-out). The 2026-04-14 user directive is explicit.
- **`additionalContext` at JSON root** in any new envelope. Phase 95-02 just fixed this; don't re-introduce.
- **Hardcoded paths to `~/.claude/plugins/mindrian-os/`** in test code or Class B/C/D/E logic. Always go through `MINDRIAN_ROOMS_HOME` (or whatever env var the planner confirms — see Open Question 1) and through the canonical `resolve-room` resolution order.
- **Running the live cascade against the user's real active room in tests.** Use `MINDRIAN_ROOMS_HOME=<scratch>` so `resolve-room` returns the fixture, not the production room. The active-room guard at `scripts/post-write` lines 207-217 is the trap that drift class C surfaces — tests must NOT trip it.
- **Crash-on-fail in any check function.** Every check must `try { ... } catch (err) { return { status: 'error', detail: err.message }; }`. The doctor run is graceful-degradation per Canon and per Phase 87 invariant.
- **Generator that overwrites user-edited ROOM.md/MINTO.md.** D-02 locks: skip if exists, `--force` overrides. Default never destroys.
- **Generator that writes ROOM.md WITHOUT the schema fields the Phase 87-01a guard requires.** The pre-commit hook only checks file presence, not content shape — but downstream tooling (`vault-section-state-generator.cjs` parsing, room-proactive skill) reads frontmatter. The hand-rolled minimal frontmatter must include at minimum `room`, `section`, `created`, `license` per the existing fixture pattern. Confirm before writing.

## Common Pitfalls

### Pitfall 1: Generator runs against dogfood `room/` BEFORE `.room-root` is committed
**What goes wrong:** The new generator walks `room/` recursively and creates 24 ROOM.md+MINTO.md files. When the user `git commit`s, the Phase 87-01a pre-commit hook fires. Without `.room-root` at `room/`, the hook's `find_room_root()` returns empty → guard SKIPS (treats as plugin source) → commit succeeds. But now the dogfood room is a Data Room with NO sentinel — exactly the Class B drift this phase is trying to close.

**Why it happens:** The generator and the sentinel are independent operations. Out-of-order execution leaves an inconsistent intermediate state.

// Source: pure Node port of scripts/post-write detect_room_section() lines 25-47
// + scripts/hooks/pre-commit-room-minto-guard.sh find_room_root() symlink safety
const fs = require('fs');
const path = require('path');

function findRoomRoot(filePath, repoRoot) {
  let dir = path.dirname(filePath);
  const visited = new Set();
  let hops = 0;
  // Realpath-normalize so symlinks don't loop or escape (R-87-01a-WIN)
  while (dir !== '/' && dir !== '.' && hops < 12) {
    let real;
    try { real = fs.realpathSync(dir); } catch (_) { return null; }
    if (visited.has(real)) break;
    visited.add(real);
    if (fs.existsSync(path.join(real, '.room-root'))) {
      const rel = path.relative(real, filePath);
      const section = rel.split(path.sep)[0];
      if (section && section !== rel) {
        return { roomDir: real, section };
      }
      return null; // file IS in the room root (not under a section)
    }
    dir = path.dirname(real);
    hops += 1;
  }
  return null;
}
```

function atomicWrite(targetPath, content) {
  const dir = path.dirname(targetPath);
  fs.mkdirSync(dir, { recursive: true });
  // Same-filesystem invariant: tmp lives in same dir as target (rename(2) is atomic)
  const tmp = path.join(dir, `.${path.basename(targetPath)}.tmp.${process.pid}`);
  fs.writeFileSync(tmp, content);
  fs.renameSync(tmp, targetPath); // atomic on POSIX
}
```

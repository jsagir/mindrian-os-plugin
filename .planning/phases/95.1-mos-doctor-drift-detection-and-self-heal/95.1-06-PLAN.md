// ── Class F: UI Ruling System compliance check ───────────────────────

const APPROVED_GLYPHS = new Set([
  '■', '▼', '▶', '▷', '✓', '•', '⚠', '⚡', '⬜', '→',
  // ├─ and └─ are tree symbols (Shape B) — multi-codepoint; handled separately
]);
const FORBIDDEN_BOX_CHARS = /[╭╮╯╰┌┐└┘│─━]/;
const FORBIDDEN_GLYPHS = /[✗✘✕]|❌|❓|❗|⚠️/;
const FORBIDDEN_ANSI = /\[(34|35|37|91|92|93|94|95|96|97)m|\\x1b\[(34|35|37|91|92|93|94|95|96|97)m|\\033\[(34|35|37|91|92|93|94|95|96|97)m/;
const FILES_EMOJI_CARVE_OUT = new Set(['scripts/context-monitor', 'context-monitor']);

const ZONE_1_RENDERER_PATTERN = /-- [\w-]+(?: -- [\w-]+)+ --/;
const ZONE_4_RENDERER_PATTERN = /▶ \/mos:/;
const RENDERER_INDICATOR = /lines\.push\(/;

function extractFrontmatterField(filePath, field) {
  let content;
  try { content = fs.readFileSync(filePath, 'utf8'); }
  catch (_) { return null; }
  const m = content.match(/^---\s*\n([\s\S]*?)\n---/m);
  if (!m) return null;
  const fm = m[1];
  const fieldRegex = new RegExp('^' + field + ':\\s*(.+)$', 'm');
  const fm_match = fm.match(fieldRegex);
  return fm_match ? fm_match[1].trim() : null;
}

function isCarveOut(filePath) {
  for (const carve of FILES_EMOJI_CARVE_OUT) {
    if (filePath.endsWith(carve)) return true;
  }
  return false;
}

function scanFileForGlyphsAndBoxChars(filePath) {
  let content;
  try { content = fs.readFileSync(filePath, 'utf8'); }
  catch (_) { return []; }
  const violations = [];
  const lines = content.split('\n');
  for (let i = 0; i < lines.length; i++) {
    const line = lines[i];
    if (FORBIDDEN_BOX_CHARS.test(line)) {
      const m = line.match(FORBIDDEN_BOX_CHARS);
      violations.push({ kind: 'box-char', char: m[0], line: i + 1, snippet: line.trim().slice(0, 80) });
    }
    if (!isCarveOut(filePath) && FORBIDDEN_GLYPHS.test(line)) {
      const m = line.match(FORBIDDEN_GLYPHS);
      violations.push({ kind: 'forbidden-glyph', char: m[0], line: i + 1, snippet: line.trim().slice(0, 80) });
    }
    if (FORBIDDEN_ANSI.test(line)) {
      const m = line.match(FORBIDDEN_ANSI);
      violations.push({ kind: 'forbidden-ansi-color', code: m[0], line: i + 1, snippet: line.trim().slice(0, 80) });
    }
  }
  return violations;
}

function scanRendererFile(filePath) {
  let content;
  try { content = fs.readFileSync(filePath, 'utf8'); }
  catch (_) { return null; }
  if (!RENDERER_INDICATOR.test(content)) return null; // not a renderer
  const issues = [];
  if (!ZONE_1_RENDERER_PATTERN.test(content)) {
    issues.push({ kind: 'missing-zone1-header', detail: 'no "-- X -- Y --" pattern found' });
  }
  if (!ZONE_4_RENDERER_PATTERN.test(content)) {
    issues.push({ kind: 'missing-zone4-footer', detail: 'no "▶ /mos:" primary action found' });
  }
  return issues.length > 0 ? issues : null;
}

function checkUIRulingCompliance(opts = {}) {
  const repoRoot = opts.repoRoot || path.resolve(__dirname, '..');
  const commandsDir = path.join(repoRoot, 'commands');
  const scriptsDir = path.join(repoRoot, 'scripts');

  const violations = {
    commandsMissingBodyShape: [],
    forbiddenBoxChars: [],
    forbiddenGlyphs: [],
    forbiddenAnsiColors: [],
    renderersMissingZones: [],
  };

  // Sub-check (a): commands/*.md missing body_shape
  if (fs.existsSync(commandsDir)) {
    let cmdFiles;
    try { cmdFiles = fs.readdirSync(commandsDir).filter(f => f.endsWith('.md')); }
    catch (_) { cmdFiles = []; }
    for (const f of cmdFiles) {
      const filePath = path.join(commandsDir, f);
      const bodyShape = extractFrontmatterField(filePath, 'body_shape');
      if (!bodyShape) {
        violations.commandsMissingBodyShape.push({ file: path.relative(repoRoot, filePath), detail: 'no body_shape field in frontmatter' });
      }
    }
  }

  // Sub-check (b) + (c): scripts/*.cjs glyph + box-char + renderer pattern scan
  if (fs.existsSync(scriptsDir)) {
    let scriptFiles;
    try { scriptFiles = fs.readdirSync(scriptsDir).filter(f => f.endsWith('.cjs')); }
    catch (_) { scriptFiles = []; }
    for (const f of scriptFiles) {
      const filePath = path.join(scriptsDir, f);
      const glyphViolations = scanFileForGlyphsAndBoxChars(filePath);
      for (const v of glyphViolations) {
        const entry = { file: path.relative(repoRoot, filePath), line: v.line, snippet: v.snippet, char: v.char || v.code };
        if (v.kind === 'box-char') violations.forbiddenBoxChars.push(entry);
        else if (v.kind === 'forbidden-glyph') violations.forbiddenGlyphs.push(entry);
        else if (v.kind === 'forbidden-ansi-color') violations.forbiddenAnsiColors.push(entry);
      }
      // Sub-check (c): zone patterns
      const rendererIssues = scanRendererFile(filePath);
      if (rendererIssues) {
        violations.renderersMissingZones.push({ file: path.relative(repoRoot, filePath), issues: rendererIssues });
      }
    }
  }

  const totalViolations =
    violations.commandsMissingBodyShape.length +
    violations.forbiddenBoxChars.length +
    violations.forbiddenGlyphs.length +
    violations.forbiddenAnsiColors.length +
    violations.renderersMissingZones.length;

  return {
    status: totalViolations === 0 ? 'ok' : 'warn',
    detail: totalViolations === 0
      ? 'all scanned files comply with UI Ruling System §1-§4'
      : totalViolations + ' violation(s) across commands/ and scripts/',
    violations,
    counts: {
      commandsMissingBodyShape: violations.commandsMissingBodyShape.length,
      forbiddenBoxChars: violations.forbiddenBoxChars.length,
      forbiddenGlyphs: violations.forbiddenGlyphs.length,
      forbiddenAnsiColors: violations.forbiddenAnsiColors.length,
      renderersMissingZones: violations.renderersMissingZones.length,
    },
    fixable: false, // D-13: --fix is detect-only in 95.1
    fixDeferredTo: '95.2 or human review',
  };
}

```javascript
const path = require('path');
const fs = require('fs');
const { McpServer } = require('@modelcontextprotocol/sdk/server/mcp.js');
const { StdioServerTransport } = require('@modelcontextprotocol/sdk/server/stdio.js');
const { z } = require('zod');

const pluginRoot = path.resolve(__dirname, '..');
const pluginMeta = require('../.claude-plugin/plugin.json');
const version = pluginMeta.version;

const server = new McpServer({ name: 'mindrian-os', version });

server.tool(
  'tool_name',
  'description',
  { arg: z.string().describe('...') },
  async ({ arg }) => {
    const result = doWork(arg);
    return { content: [{ type: 'text', text: JSON.stringify(result) }] };
  }
);

async function main() {
  const transport = new StdioServerTransport();
  await server.connect(transport);
  process.stderr.write(`[mindrian-brain] MCP server v${version} started (stdio)\n`);
}
main().catch((err) => {
  process.stderr.write(`[mindrian-brain] Fatal: ${err.message}\n`);
  process.exit(1);
});
```

```
{
  packet_version: "1.0",
  packet_type: "DirectiveEnvelope",
  mode: "GUIDED" | "AUTONOMOUS" | "HYBRID",           // CANONICAL DEFAULT: GUIDED
  mode_rationale: "<short why-this-mode string>",
  directive: { guided?: {...}, autonomous?: {...}, hybrid?: {...} },
  user_override: {

    "let me think": "switch to GUIDED, ask the next reframing question",
    "stop": "halt, return control"
  },

}
```

Mode-selection signals (Brain reads from local context; the shim's wrapDirective applies the same rules CLIENT-SIDE when the Brain response lacks a mode):
- User just dropped material with no stated question -> GUIDED
- User explicitly said "run the analysis on X" -> AUTONOMOUS
- User in-room 8+ sessions, mature, commit phase -> HYBRID
- User brand new, first material -> GUIDED (NEVER autonomous on cold start)
- User said "just tell me" in last 2 turns -> AUTONOMOUS

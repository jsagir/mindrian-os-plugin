    const path = require('node:path');
    const fs = require('node:fs');
    const { spawn } = require('node:child_process');

    const LAYERS = Object.freeze([
      { id: 'plugin_root',      name: 'L1 plugin-root-resolver' },
      { id: 'key_resolver',     name: 'L2 brain-key-resolver' },
      { id: 'https_schema',     name: 'L3 HTTPS schema probe' },
      { id: 'stdio_handshake',  name: 'L4 MCP stdio handshake' },
      { id: 'e2e_brain_schema', name: 'L5 e2e brain_schema via shim' },

    const STDIO_TIMEOUT_MS = Number(process.env.MINDRIAN_BRAIN_SMOKE_TIMEOUT_MS) || 10000;
    const OVERALL_BUDGET_MS = 30000;

    function _now() { return Date.now(); }

    async function _runLayer(name, fn) {
      const t0 = _now();
      try {
        const r = await fn();
        const ms = _now() - t0;
        return { ok: !!r.ok, reason: r.reason || (r.ok ? 'pass' : 'fail'), ms };
      } catch (e) {
        return { ok: false, reason: 'exception: ' + (e && e.message ? e.message : String(e)), ms: _now() - t0 };

    // The real stdio orchestrator. opts.intent is 'handshake' (L4) or
    // 'e2e_brain_schema' (L5). Wraps spawn + JSON-RPC write/read + timeout.
    function _spawnAndHandshake(shimPath, opts) {
      return new Promise((resolve) => {
        const proc = spawn(process.execPath, [shimPath], {
          stdio: ['pipe', 'pipe', 'pipe'],
          env: process.env,
        });
        const timer = setTimeout(() => {
          try { proc.kill('SIGTERM'); } catch (_) {}
          resolve({ ok: false, reason: opts.intent + ' timed out after ' + STDIO_TIMEOUT_MS + 'ms' });
        }, STDIO_TIMEOUT_MS);
        let buf = '';
        proc.stdout.on('data', (chunk) => {
          buf += chunk.toString('utf8');
          // MCP frames are newline-delimited JSON. Parse each complete line.
          const lines = buf.split('\n');
          buf = lines.pop();
          for (const line of lines) {
            if (!line.trim()) continue;
            try {
              const msg = JSON.parse(line);
              if (msg.id === 1 && msg.result && msg.result.serverInfo) {
                if (opts.intent === 'handshake') {
                  clearTimeout(timer); proc.kill('SIGTERM');
                  return resolve(msg.result.serverInfo.name === 'mindrian-brain'
                    ? { ok: true, reason: 'handshake succeeded, server=mindrian-brain v' + msg.result.serverInfo.version }
                    : { ok: false, reason: 'unexpected serverInfo.name=' + msg.result.serverInfo.name });
                }
                // L5: after initialize succeeds, send tools/call brain_schema
                proc.stdin.write(JSON.stringify({
                  jsonrpc: '2.0', id: 2, method: 'tools/call',
                  params: { name: 'brain_schema', arguments: {} },
                }) + '\n');
              } else if (msg.id === 2 && msg.result && msg.result.content) {
                clearTimeout(timer); proc.kill('SIGTERM');

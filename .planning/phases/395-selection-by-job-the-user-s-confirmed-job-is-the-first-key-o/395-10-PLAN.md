* **Compose rule** (Theo handoffs of 2026-11-16): `pws_jtbd` gives `{ status: 'ruled', main: true }`; each `pws_jtbd_extra` item gives `{ status: 'ruled', main: false }`; a draft whose text equals a ruled text is skipped; each other `pws_jtbd_drafts[i]` gives `{ status: 'draft', main: i === 0 and no pws_jtbd, confidence, drafted_by }`. `pws_jtbd_basis`, when present, goes on each ruled statement of that node as `basis`. // jobsPage(res) -- one query answer to { kind: 'rows', rows } | { kind: 'cap' } | { kind: 'fail', reason }. A page of
// JOBS_PAGE_CAP rows or more counts as capped (LIMIT 100 can cut a bucket without a notice).
function jobsPage(res) {
  if (!res || typeof res !== 'object') return { kind: 'fail', reason: 'no answer' };
  const rows = Array.isArray(res.rows) ? res.rows : Array.isArray(res.records) ? res.records : null;
  if (typeof res.text === 'string' && /ROW_CAP/.test(res.text) && (!rows || rows.length === 0)) return { kind: 'cap' };
  if (!rows || res.error || res.degraded) return { kind: 'fail', reason: typeof res.error === 'string' ? res.error : 'no rows array' };
  if (rows.length >= JOBS_PAGE_CAP) return { kind: 'cap' };
  return { kind: 'rows', rows };
}

function hasItems(v) {

// pullJobRows(query) -- the bucketed job statements read. query is brain-client's query(cypher, params) or a test
// stub. Returns { ok: true, rows, counts } only when every bucket answered and the rows equal Theo's own count;
// else { ok: false, reason }. Never writes.
async function pullJobRows(query) {
  if (typeof query !== 'function') return { ok: false, reason: 'no query function' };
  const rows = [];
  async function ask(cypher, params) {
    try {
      return jobsPage(await query(cypher, params));
    } catch (_e) {
      return { kind: 'fail', reason: 'the read threw' };
    }
  }
  async function bucket(prefix, depth) {
    const p = await ask(CYPHER_JOBS_PREFIX, { n: prefix.length, prefix: prefix });
    if (p.kind === 'rows') {
      rows.push.apply(rows, p.rows);
      return null;
    }
    if (p.kind === 'fail') return 'bucket "' + prefix + '": ' + p.reason;
    if (depth >= JOBS_MAX_DEPTH) return 'bucket "' + prefix + '" is still capped at depth ' + depth;
    for (const c of JOBS_ALNUM) {
      const bad = await bucket(prefix + c, depth + 1);
      if (bad) return bad;
    }
    return null;
  }
  for (const c of JOBS_ALNUM) {
    const bad = await bucket(c, 0);
    if (bad) return { ok: false, reason: bad };
  }
  const other = await ask(CYPHER_JOBS_OTHER, { alnum: JOBS_ALNUM });
  if (other.kind !== 'rows') return { ok: false, reason: 'catch-all bucket: ' + (other.kind === 'cap' ? 'capped' : other.reason) };
  rows.push.apply(rows, other.rows);

function rowMarkErrors(rows) {

/**
 * methodJobForCommand(command) -- the one job statement shown beside a proposed command: the ruled main statement of
 * the first framework of the command (data/command-registry.json order) that has one, else the main draft of the
 * first framework that has a draft. Display only. A ruled statement with a basis mark that names a model keeps the
 * mark and says that a model drafted it. A statement with pws_backed true also carries pws_backed and the backing
 * sentence; a statement with pws_backed false carries neither (no claim). Returns null when there is none.
 * @returns {{ framework, alias_of?, statement, status: 'ruled'|'draft', basis?, confidence?, note, pws_backed?: true, backing? }|null}
 */
function methodJobForCommand(command) {

  const ruled = found.find((x) => x.j.ruled_main);
  if (ruled) {
    const b = ruled.j.ruled_basis;
    if (b) return Object.assign(head(ruled), { statement: ruled.j.ruled_main, status: 'ruled', basis: b, note: /model/i.test(b) ? RULED_BY_RULE_NOTE : RULED_NOTE }, backing(ruled.j.ruled_main_pws_backed));
    return Object.assign(head(ruled), { statement: ruled.j.ruled_main, status: 'ruled', note: RULED_NOTE }, backing(ruled.j.ruled_main_pws_backed));
  }
  const drafted = found.find((x) => x.j.drafts.length > 0);
  if (drafted) {
    const d = drafted.j.drafts.find((x) => x.main) || drafted.j.drafts[0];
    return Object.assign(head(drafted), { statement: d.text, status: 'draft', confidence: d.confidence, note: DRAFT_NOTE }, backing(d.pws_backed));
  }
  return null;
}

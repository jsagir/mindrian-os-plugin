## 1. Decisions of this plan

* **Storage: a separate file `data/framework-jobs.json` with its own hash.** Reasons (READ):
  1. `tests/test-355-framework-names.cjs` Leg C pins the exact top-level keys of `buildSnapshot`; a new key in `data/framework-names.json` fails it.
  2. `ui/mindrian-workspace-mod/scripts/sync-assets.cjs` copies `data/framework-names.json` into the browser workspace assets; statements would go to a surface that this plan does not review.
  3. `scripts/release-lib/canon-snapshot-gate.sh` and `lib/core/part8-egress-guard.cjs` read `data/framework-names.json` only; they stay unchanged and their hash stays a hash of names.
  4. `package.json` `files` lists `data` as a whole folder (only 3 named data files are excluded), so the new file ships in the npm tarball and in the Desktop copy (npm pack minus `bin/`) with no change. No `.npmignore` exists.
* **Door:** `--live` reads the statements with 3 fixed Cypher texts through `brainClient.query` (36 first-character buckets, split on a cap up to 3 characters, 1 catch-all bucket, 1 count read). The rows must equal the count of Theo (selected, ruled, with_extra, with_drafts). Any failure: exit 4, NEITHER file is written. The jobs read runs only on the real wire path (no injected `askOp`), the same rule as `readMappedBy`, so every existing test of `runLive` keeps its state.
* **Compose rule** (Theo handoffs of 2026-11-16): `pws_jtbd` gives `{ status: 'ruled', main: true }`; each `pws_jtbd_extra` item gives `{ status: 'ruled', main: false }`; a draft whose text equals a ruled text is skipped; each other `pws_jtbd_drafts[i]` gives `{ status: 'draft', main: i === 0 and no pws_jtbd, confidence, drafted_by }`. `pws_jtbd_basis`, when present, goes on each ruled statement of that node as `basis`.
* **Allowed keys (closed lists, checked by `--check` and by the reader):** top level `snapshot_note, snapshot_date, source, source_sha256, counts, frameworks`; entry `name, alias_of, jobs`; ruled job `text, status, main` (optional `basis`); draft job `text, status, main, confidence, drafted_by`. Status set `ruled, draft`. Confidence set `high, medium, low, unknown`. No em dash or en dash anywhere.
* **Where a statement is shown: one place only.** The `suggest_next` reply gets `next_move.method_job` (omitted when there is none): the statement of the framework behind the PRIMARY command (`data/command-registry.json` `frameworks`), with `status`, a `note`, and `basis` or `confidence`. A ruled statement first; else the main draft. Not shown in this plan: the brief (`BRIEF.md` is a room file with token-budget plans and a record-basis fingerprint; a new line changes the plan choice and needs the jobs file as a basis input), the per-turn hook block (its files belong to plan 395-08), the browser workspace.
* **Not steering:** no sort key, score, ledger build or filter reads the file. Proof by test F4: on every nest of a born room, `nextMoveForSection` output (both surfaces) and the whole `suggest_next` reply are byte-identical with and without the file, except the `method_job` field. Test F5 checks that 7 selection files do not name `framework-jobs`.
* **Wording a user's model reads** (`lib/core/framework-jobs.cjs`):
  * ruled, no basis: "The job this method serves, as the owner of the method approved it. Shown for reference only; it does not rank the move."
  * ruled, basis names a model: "The job this method serves. A model drafted it and the owner of the method approved it by a rule, not word by word. Shown for reference only; it does not rank the move."
  * draft: "Draft: a model wrote this job statement and the owner of the method has not confirmed it yet. Say that it is a draft; do not present it as the method's own statement. It does not rank the move."
* **Alias twins:** resolved offline, one step: an entry with no statement and an `alias_of` reads the statements of its target. The alias link is a framework name (names already ship). Limit: an alias of an alias is not followed.

const JOBS_RETURN = [
  "RETURN f.name AS name, toString(coalesce(f.alias_of, '')) AS alias_of,",
  "toString(coalesce(f.pws_jtbd, '')) AS pws_jtbd, coalesce(f.pws_jtbd_extra, []) AS pws_jtbd_extra,",
  "coalesce(f.pws_jtbd_drafts, []) AS pws_jtbd_drafts, coalesce(f.pws_jtbd_draft_confidence, '') AS pws_jtbd_draft_confidence,",
  "coalesce(f.pws_jtbd_draft_by, '') AS pws_jtbd_draft_by, toString(coalesce(f.pws_jtbd_basis, '')) AS pws_jtbd_basis",

const CYPHER_JOBS_PREFIX = ['MATCH (f:Framework)', 'WHERE toLower(left(f.name, $n)) = $prefix', 'AND', JOBS_HAS, JOBS_RETURN].join(' ');
const CYPHER_JOBS_OTHER = ['MATCH (f:Framework)', 'WHERE NOT toLower(left(f.name, 1)) IN $alnum', 'AND', JOBS_HAS, JOBS_RETURN].join(' ');
const CYPHER_JOBS_COUNT = [
  'MATCH (f:Framework)',
  'WHERE',
  JOBS_HAS,
  'RETURN count(f) AS selected,',
  "sum(CASE WHEN toString(coalesce(f.pws_jtbd, '')) <> '' THEN 1 ELSE 0 END) AS ruled,",
  'sum(CASE WHEN size(coalesce(f.pws_jtbd_extra, [])) > 0 THEN 1 ELSE 0 END) AS with_extra,',

const JOBS_SNAPSHOT_NOTE =

const JOBS_ALNUM = '0123456789abcdefghijklmnopqrstuvwxyz'.split('');
const JOBS_PAGE_CAP = 100;
const JOBS_MAX_DEPTH = 2;

// jobsPage(res) -- one query answer to { kind: 'rows', rows } | { kind: 'cap' } | { kind: 'fail', reason }. A page of
// JOBS_PAGE_CAP rows or more counts as capped (LIMIT 100 can cut a bucket without a notice).
function jobsPage(res) {
  if (!res || typeof res !== 'object') return { kind: 'fail', reason: 'no answer' };
  const rows = Array.isArray(res.rows) ? res.rows : Array.isArray(res.records) ? res.records : null;
  if (typeof res.text === 'string' && /ROW_CAP/.test(res.text) && (!rows || rows.length === 0)) return { kind: 'cap' };
  if (!rows || res.error || res.degraded) return { kind: 'fail', reason: typeof res.error === 'string' ? res.error : 'no rows array' };
  if (rows.length >= JOBS_PAGE_CAP) return { kind: 'cap' };
  return { kind: 'rows', rows };
}

function hasItems(v) {
  return (Array.isArray(v) || typeof v === 'string') && v.length > 0;
}

// pullJobRows(query) -- the bucketed job statements read. query is brain-client's query(cypher, params) or a test
// stub. Returns { ok: true, rows, counts } only when every bucket answered and the rows equal Theo's own count;
// else { ok: false, reason }. Never writes.
async function pullJobRows(query) {
  if (typeof query !== 'function') return { ok: false, reason: 'no query function' };
  const rows = [];
  async function ask(cypher, params) {
    try {
      return jobsPage(await query(cypher, params));
    } catch (_e) {
      return { kind: 'fail', reason: 'the read threw' };
    }
  }
  async function bucket(prefix, depth) {
    const p = await ask(CYPHER_JOBS_PREFIX, { n: prefix.length, prefix: prefix });
    if (p.kind === 'rows') {
      rows.push.apply(rows, p.rows);
      return null;
    }
    if (p.kind === 'fail') return 'bucket "' + prefix + '": ' + p.reason;
    if (depth >= JOBS_MAX_DEPTH) return 'bucket "' + prefix + '" is still capped at depth ' + depth;
    for (const c of JOBS_ALNUM) {
      const bad = await bucket(prefix + c, depth + 1);
      if (bad) return bad;
    }
    return null;
  }
  for (const c of JOBS_ALNUM) {
    const bad = await bucket(c, 0);
    if (bad) return { ok: false, reason: bad };
  }
  const other = await ask(CYPHER_JOBS_OTHER, { alnum: JOBS_ALNUM });
  if (other.kind !== 'rows') return { ok: false, reason: 'catch-all bucket: ' + (other.kind === 'cap' ? 'capped' : other.reason) };
  rows.push.apply(rows, other.rows);
  const counted = await ask(CYPHER_JOBS_COUNT, {});
  const want = counted.kind === 'rows' ? counted.rows[0] : null;
  if (!want || typeof want.selected !== 'number') return { ok: false, reason: 'the count read failed' };
  const got = {
    selected: rows.length,
    ruled: rows.filter((r) => r && typeof r.pws_jtbd === 'string' && r.pws_jtbd !== '').length,
    with_extra: rows.filter((r) => r && hasItems(r.pws_jtbd_extra)).length,
    with_drafts: rows.filter((r) => r && hasItems(r.pws_jtbd_drafts)).length,
  };
  for (const k of ['selected', 'ruled', 'with_extra', 'with_drafts']) {
    if (got[k] !== want[k]) {
      return { ok: false, reason: 'the rows do not match the count of Theo (' + k + ': read ' + got[k] + ', counted ' + want[k] + ')' };
    }
  }
  return { ok: true, rows: rows, counts: got };
}

  return require(path.join(REPO_ROOT, 'lib', 'core', 'brain-client.cjs')).query;
}

function clone(v) { return JSON.parse(JSON.stringify(v)); }
function manyNames(n) {
  const out = [];
  for (let i = 0; i < n; i++) out.push({ name: 'Framework ' + String(i).padStart(4, '0') });
  return out;
}
// A stub of brain-client query over the given rows: answers the three fixed Cypher texts of the refresh script.
function stubQuery(rows, opts) {
  const o = opts || {};
  const calls = [];
  const fn = async (cypher, params) => {
    calls.push({ cypher, params });
    if (cypher === refresh.CYPHER_JOBS_COUNT) {
      if (o.countFails) return null;
      const r = rows;
      return { rows: [{
        selected: r.length + (o.countOff || 0),
        ruled: r.filter((x) => x.pws_jtbd !== '').length,
        with_extra: r.filter((x) => x.pws_jtbd_extra.length > 0).length,
        with_drafts: r.filter((x) => x.pws_jtbd_drafts.length > 0).length,
      }] };
    }
    if (cypher === refresh.CYPHER_JOBS_OTHER) {
      return { rows: rows.filter((x) => params.alnum.indexOf(x.name.slice(0, 1).toLowerCase()) === -1) };
    }
    if (cypher === refresh.CYPHER_JOBS_PREFIX) {
      if (o.failPrefix === params.prefix) return null;
      if (o.capPrefix && params.prefix.indexOf(o.capPrefix) === 0 && params.prefix.length <= (o.capDepth || 1)) return { text: 'ROW_CAP: too many rows' };
      return { rows: rows.filter((x) => x.name.slice(0, params.n).toLowerCase() === params.prefix) };
    }
    return { error: 'unexpected cypher' };
  };
  fn.calls = calls;
  return fn;
}

function fakeServer() {
  const registered = [];

    server: {
      tool(name, _d, a, b) { registered.push({ name, handler: typeof b === 'function' ? b : a }); },
      registerTool(name, _c, handler) { registered.push({ name, handler }); },
      server: { getClientCapabilities() { return {}; } },
    },
  };
}

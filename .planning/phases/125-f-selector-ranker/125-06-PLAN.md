```javascript
'use strict';
// Phase 125-06 -- F-selector decision recording + decay weight (D7 implementation).
// All writes route through navigation.cjs chokepoint (Plan 00 writeEdge + Phase 109
// logMemoryEvent + Phase 109 findRecentChanges). Canon Part 4 (every choice is
// graph data) + Canon Part 7 (reuse the closed surface) + Canon Part 8 (LOCAL only;
// no Brain calls; no user content leaves the room).

const navigation = require('../core/navigation.cjs');

// Decay window N per D7 + CONTEXT.md Open Question #6 (default 5; config override
// in v2 per .mos/config.json).
const DECAY_WINDOW = 5;
// Exclusion threshold per RESEARCH G-09 (filter from top-K when decay factor
// is below this; freshly-rejected commands shouldn't appear in the very next
// selector). Configurable in v2.
const EXCLUSION_THRESHOLD = 0.1;
// Default DEFERRED edge clock-window per CONTEXT.md Open Question #7 (30 days
// from decision; clock-driven expiry alongside the invocation-count decay).
const DEFAULT_DEFER_EXPIRY_DAYS = 30;

function _defaultExpiresAt() {
  return new Date(Date.now() + DEFAULT_DEFER_EXPIRY_DAYS * 24 * 3600 * 1000).toISOString();
}

function recordSelectorDecision(args) {
  const o = args || {};
  const db = o.db;
  const decision = o.decision;
  const command = o.command;
  const framework = o.framework;
  const reason = (typeof o.reason === 'string' && o.reason.length > 0) ? o.reason : null;
  const roomState = o.roomState || {};
  const score_at_decision = (typeof o.score_at_decision === 'number') ? o.score_at_decision : null;

  if (decision !== 'defer' && decision !== 'reject') {
    return { ok: false, reason: 'invalid_decision' };
  }
  if (typeof command !== 'string' || command.length === 0) {
    return { ok: false, reason: 'invalid_command' };
  }
  if (typeof framework !== 'string' || framework.length === 0) {
    return { ok: false, reason: 'invalid_framework' };
  }
  if (!db) {
    return { ok: false, reason: 'invalid_db' };
  }

  const investment_level_at_decision = (typeof roomState.investment_level === 'number') ? roomState.investment_level : 0;
  const edgeSemantic = (decision === 'defer') ? 'DEFERRED' : 'REJECTED';
  const expiresAt = (decision === 'defer') ? _defaultExpiresAt() : null;

  // Step 1: write memory_event via navigation chokepoint.
  const memEvent = navigation.logMemoryEvent(db, 'f_selector_decision', {
    decision,
    command,
    framework,
    reason,
    edge_semantic: edgeSemantic,
    expires_at: expiresAt,
    score_at_decision,
    investment_level_at_decision,
    source_path: 'f-selector:' + command,
    created_by: 'user',
  });
  if (!memEvent || !memEvent.ok) {
    return { ok: false, reason: 'memory_event_failed', detail: memEvent ? memEvent.reason : 'unknown' };
  }
  const decision_id = memEvent.eventId;

  // Step 2: write typed cascade edge via navigation.writeEdge (Plan 00).
  const edgeProps = (decision === 'defer')
    ? { reason, decision_id, expires_at: expiresAt }
    : { reason, decision_id };
  const edgeResult = navigation.writeEdge({
    db,
    source_id: 'cmd:' + command,
    target_id: 'framework:' + framework,
    edge_type: edgeSemantic,
    properties: edgeProps,
  });
  // navigation.writeEdge signature is writeEdge(db, params) in some implementations
  // -- if the Plan 00 signature is (params) with db inside params, pass db as part of
  // params; if the signature is (db, params), pass db first. The Plan 00 spec shows
  // db is the first positional arg via the openRoomDb-style pattern. Verify and adjust.
  // (The actual implementation from Plan 00 takes (db, params); fix invocation site here
  //  if Plan 00's final signature differs.)

  if (!edgeResult || !edgeResult.ok) {
    return { ok: false, reason: 'edge_write_failed', detail: edgeResult ? edgeResult.reason : 'unknown' };
  }

  return {
    ok: true,
    decision_id,
    edge_id: edgeResult.edge_id,
  };
}

// Count invocations since the last f_selector_decision for this command.
// Reads via navigation.findRecentChanges (no direct room-db access). The
// "invocations since" proxy: count of framework_invoked events newer than
// the decision's created_at. When roomState provides a pre-computed counter
// for testing, prefer that.
function _invocationsSinceDecision(db, command, roomState) {
  if (roomState && roomState.invocationsSinceDecision && typeof roomState.invocationsSinceDecision[command] === 'number') {
    return roomState.invocationsSinceDecision[command];
  }
  if (!db) return 0;
  // Find most recent f_selector_decision for this command.
  const recent = navigation.findRecentChanges(db, 0, { eventType: 'f_selector_decision', limit: 100 });
  let lastDecisionAt = null;
  for (const r of recent) {
    if (r.properties && r.properties.command === command) {
      if (lastDecisionAt === null || r.createdAt > lastDecisionAt) {
        lastDecisionAt = r.createdAt;
      }
    }
  }
  if (lastDecisionAt === null) return Infinity; // no decision -> no decay
  // Count framework_invoked events newer than lastDecisionAt.
  const invocations = navigation.findRecentChanges(db, lastDecisionAt, { eventType: 'framework_invoked', limit: 1000 });
  return invocations.length;
}

// D7 exponential decay. Pure when roomState supplies the counter.
function applyDecayWeight(base_score, command_id, roomState) {
  if (typeof base_score !== 'number' || !isFinite(base_score)) return 0;
  const db = (roomState && roomState.db) ? roomState.db : null;
  const n = _invocationsSinceDecision(db, command_id, roomState);
  if (n === Infinity) return base_score; // no decision recorded; no decay
  const factor = 1 - Math.exp(-(n / DECAY_WINDOW));
  return base_score * factor;
}

// RESEARCH G-09: filter helper. Plan 05 ranker may wire this to exclude
// freshly-rejected commands from top-K.
function shouldExclude(command_id, roomState) {
  const db = (roomState && roomState.db) ? roomState.db : null;
  const n = _invocationsSinceDecision(db, command_id, roomState);
  if (n === Infinity) return false;
  const factor = 1 - Math.exp(-(n / DECAY_WINDOW));
  return factor < EXCLUSION_THRESHOLD;
}

module.exports = {
  recordSelectorDecision,
  applyDecayWeight,
  shouldExclude,
  DECAY_WINDOW,
  EXCLUSION_THRESHOLD,
  DEFAULT_DEFER_EXPIRY_DAYS,
  // Test seam.
  _test: { _invocationsSinceDecision, _defaultExpiresAt },
};
```

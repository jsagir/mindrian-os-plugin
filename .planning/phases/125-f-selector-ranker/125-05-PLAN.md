'use strict';
// Phase 125-05 -- F-selector top-K ranker. Pure synchronous function.
// CONTEXT.md D1-D11 implementation. Plan 05 of the 9-plan F-selector ranker phase.
//
// HARD PRECONDITION: Phase 104.1 ships jtbd_label + jtbd_summary + teaching fields
// in data/command-registry.json. The ranker fails closed on every command if
// Phase 104.1 hasn't shipped (returns empty list, no crash).
//
// Canon Part 9: SQL navigates; this ranker is the "ranks structured packets" face.
// Canon Part 7: extends shipped command-resolver + chain-recommender + packet
//               without modifying their closed surfaces.
// Canon Part 8: zero Brain calls (consumes packet, never issues query); LOCAL only.

const path = require('node:path');
const fs = require('node:fs');
const projections = require('../core/navigation/projections.cjs');

const REGISTRY_PATH = path.join(__dirname, '..', '..', 'data', 'command-registry.json');
const TAXONOMY_PATH = path.join(__dirname, '..', 'hmi', 'jtbd-taxonomy.json');
const DEFAULT_SEED = 'Beautiful Question Framework';

let _registryCache = null;
let _taxonomyCache = null;

function _loadRegistry() {
  if (_registryCache) return _registryCache;
  try { _registryCache = JSON.parse(fs.readFileSync(REGISTRY_PATH, 'utf8')); }
  catch (_e) { _registryCache = { commands: [] }; }
  return _registryCache;
}
function _loadTaxonomy() {
  if (_taxonomyCache) return _taxonomyCache;
  try { _taxonomyCache = JSON.parse(fs.readFileSync(TAXONOMY_PATH, 'utf8')); }
  catch (_e) { _taxonomyCache = { entries: [] }; }
  return _taxonomyCache;
}

// Test seam: reset caches between tests.
function _resetCaches() { _registryCache = null; _taxonomyCache = null; }

// D9 implementation. Pure. Same inputs -> same output.
function selectWhyContent(jtbd_summary, teaching, investment_level) {
  const hasJtbd = typeof jtbd_summary === 'string' && jtbd_summary.length > 0;
  const hasTeaching = typeof teaching === 'string' && teaching.length > 0;
  if (!hasJtbd && !hasTeaching) return '';
  if (!hasJtbd) return teaching;
  if (!hasTeaching) return jtbd_summary;
  const lvl = (typeof investment_level === 'number') ? investment_level : 0;
  if (lvl < 0.4) return teaching;
  if (lvl >= 0.7) return jtbd_summary;
  // Mid-band: stitch teaching + " -- " + jtbd_summary (double-hyphen per the
  // project no-em-dash rule + CONTEXT.md D9 spec).
  return teaching + ' -- ' + jtbd_summary;
}

// D5 visible-investment badge.
function renderInvestmentBadge(investment_level) {
  const lvl = (typeof investment_level === 'number') ? Math.max(0, Math.min(1, investment_level)) : 0;
  const pct = Math.round(lvl * 100);
  const remaining = Math.max(0, 10 - Math.round(lvl * 10));
  if (lvl === 0) return 'Investment: 0% -- Brain priors only (10 invocations until full local)';
  if (lvl >= 1.0) return 'Investment: 100% -- full local scoring with Brain confidence';
  return 'Investment: ' + pct + '% -- Brain + local signal (' + remaining + ' invocations to full)';
}

// D5 slice rationale badge.
function renderSliceBadge(slice_scope, slice_rationale) {
  const scope = (slice_scope === 1 || slice_scope === 2 || slice_scope === 3) ? slice_scope : 3;
  const rationaleShort = (typeof slice_rationale === 'string' ? slice_rationale : '').slice(0, 50);
  return 'Slice: ' + scope + ' hop' + (scope === 1 ? '' : 's') + ' -- ' + rationaleShort;
}

// D7 decay-weight integration. When opts.applyDecayWeight is provided (typically
// by the F-selector renderer that wires Plan 05 + Plan 06 together), apply it.
// When absent, default no-op.
function _applyDecay(applyDecayWeight, baseScore, commandId, roomState) {
  if (typeof applyDecayWeight !== 'function') return baseScore;
  try {
    const adjusted = applyDecayWeight(baseScore, commandId, roomState);
    return (typeof adjusted === 'number' && isFinite(adjusted)) ? adjusted : baseScore;
  } catch (_e) { return baseScore; }
}

// D9 + JTBD selection per RESEARCH G-01.
function _resolveJtbdForCommand(cmd, roomState) {
  const serves = Array.isArray(cmd.serves_jtbd) ? cmd.serves_jtbd : [];
  if (serves.length === 0) return null;
  const activeJtbd = (roomState && typeof roomState.activeJtbd === 'string') ? roomState.activeJtbd : null;
  if (activeJtbd && serves.indexOf(activeJtbd) !== -1) return activeJtbd;
  return serves[0];
}

// Extract brain_confidence for this command's framework from the packet's chain hint.
function _brainConfidenceFromPacket(packetOptional, framework) {
  if (!packetOptional || !packetOptional.local_graph_summary) return null;
  const hint = packetOptional.local_graph_summary.framework_chain_hint;
  if (!hint || !Array.isArray(hint.edges)) return null;
  // Highest confidence edge that targets this framework.
  let best = null;
  for (const e of hint.edges) {
    if ((e.from === framework || e.to === framework) && typeof e.confidence === 'number') {
      if (best === null || e.confidence > best) best = e.confidence;
    }
  }
  return best;
}

// recency_decay for D4: 1.0 = fully decayed (old); 0.0 = fresh.
// Computed from roomState.lastInvokedAt[command] if available, else 0.5 default.
function _recencyDecay(cmd, roomState) {
  if (!roomState || !roomState.lastInvokedAt) return 0.5;
  const last = roomState.lastInvokedAt[cmd.command];
  if (typeof last !== 'number') return 0.5;
  const ageDays = (Date.now() - last) / (24 * 3600 * 1000);
  return Math.min(1.0, ageDays / 30); // 30 days -> 1.0 (fully decayed)
}

// problem_type_bind for D4: 1.0 if command's frameworks include one the
// roomState.problemType prefers; 0.0 otherwise.
function _problemTypeBind(cmd, roomState) {
  if (!roomState || typeof roomState.problemType !== 'string') return 0.5;
  const pt = roomState.problemType.toUpperCase();
  const fws = Array.isArray(cmd.frameworks) ? cmd.frameworks : [];
  // Simple proxy: methodology commands match WDP; exploration commands match UDP/IDP.
  if (pt === 'WDP' && cmd.kind === 'methodology') return 1.0;
  if ((pt === 'UDP' || pt === 'IDP') && fws.indexOf('Beautiful Question Framework') !== -1) return 1.0;
  return 0.5;
}

// D4 scoring formula.
function _scoreCommand({ cmd, packetOptional, roomState, investment_level }) {
  const fw = (Array.isArray(cmd.frameworks) && cmd.frameworks.length > 0) ? cmd.frameworks[0] : DEFAULT_SEED;
  const bc = _brainConfidenceFromPacket(packetOptional, fw);
  const brain_confidence = (typeof bc === 'number') ? bc : 0.5; // default mid when no packet signal
  const recency_decay = _recencyDecay(cmd, roomState);
  const problem_type_bind = _problemTypeBind(cmd, roomState);
  const inv = investment_level;
  const numerator = brain_confidence * 0.40
    + (1 - recency_decay) * 0.30 * inv
    + problem_type_bind * 0.30 * inv;
  const denominator = 0.40 + 0.30 * inv + 0.30 * inv;
  return numerator / denominator;
}

function _sourceFor(packetOptional, cmd) {
  if (packetOptional && packetOptional.local_graph_summary && packetOptional.local_graph_summary.framework_chain_hint) {
    const hint = packetOptional.local_graph_summary.framework_chain_hint;
    if (Array.isArray(hint.edges) && hint.edges.length > 0) return 'packet';
  }
  if (Array.isArray(cmd.frameworks) && cmd.frameworks.length > 0) return 'chain';
  return 'registry-only';
}

// MAIN SIGNATURE per CONTEXT.md.
function rankForSelector(args) {
  const o = args || {};
  const jtbd = o.jtbd;
  const problemType = o.problemType;
  const focusNodeId = o.focusNodeId;
  const roomState = o.roomState || {};
  const packetOptional = o.packetOptional || null;
  const k = (typeof o.k === 'number' && o.k > 0) ? Math.floor(o.k) : 3;
  const applyDecayWeight = (o._applyDecayWeight && typeof o._applyDecayWeight === 'function') ? o._applyDecayWeight : null;

  // Snapshot investment level at rank time (D9 invariant).
  const { level: investment_level } = projections.computeInvestmentLevel(roomState);

  const reg = _loadRegistry();
  const commands = Array.isArray(reg.commands) ? reg.commands : [];
  const scored = [];
  for (const cmd of commands) {
    if (!cmd || typeof cmd.command !== 'string') continue;
    // D6 fail-closed: jtbd_summary required.
    const jtbd_summary = (typeof cmd.jtbd_summary === 'string' && cmd.jtbd_summary.length > 0) ? cmd.jtbd_summary : null;
    if (jtbd_summary === null) continue;
    // D11 fail-closed: teaching required.
    const teaching = (typeof cmd.teaching === 'string' && cmd.teaching.length > 0) ? cmd.teaching : null;
    if (teaching === null) continue;
    const jtbd_label = (typeof cmd.jtbd_label === 'string') ? cmd.jtbd_label : '';
    const framework = (Array.isArray(cmd.frameworks) && cmd.frameworks.length > 0) ? cmd.frameworks[0] : DEFAULT_SEED;
    const baseScore = _scoreCommand({ cmd, packetOptional, roomState, investment_level });
    const adjustedScore = _applyDecay(applyDecayWeight, baseScore, cmd.command, roomState);
    const why = selectWhyContent(jtbd_summary, teaching, investment_level);
    const source = _sourceFor(packetOptional, cmd);
    scored.push({
      command: cmd.command,
      jtbd_label,
      jtbd_summary,
      teaching,
      framework,
      score: Math.max(0, Math.min(1, adjustedScore)),
      why,
      source,
      investment_level,
    });
  }
  // Sort score desc.
  scored.sort((a, b) => b.score - a.score);
  // Tier 0 / cold-start fallback: if zero eligible commands but k > 0, return at
  // minimum 1 default-seed-mapped command per RESEARCH G-08. This depends on
  // the registry having at least one command with content; if Phase 104.1 has
  // shipped, this never triggers (lots of eligible commands).
  if (scored.length === 0) {
    return [];
  }
  return scored.slice(0, k);
}

  // Test seam.
  _test: { _resetCaches, _scoreCommand, _brainConfidenceFromPacket, _resolveJtbdForCommand },
};

```javascript
// Phase 125-03 -- stitch helper. Reads roomState via projections, fetches slice
// via Plan 02 fetcher (or mock), returns the hint shape OR undefined when the
// active set is empty (which is the "absent from packet" path -- see Plan 04
// schema: required: [] for framework_chain_hint).
async function _surfaceFrameworkChainHint(db, roomState, opts) {
  if (!roomState || typeof roomState !== 'object') return undefined;
  const active = projections.resolveActiveFrameworks(roomState);
  if (!Array.isArray(active) || active.length === 0) return undefined;
  const { depth } = projections.resolveHopDepth(roomState);
  const activeNames = active.map((a) => a.name);
  const fetcher = (opts && opts._mocks && typeof opts._mocks.fetchFrameworkChainSlice === 'function')
    ? opts._mocks.fetchFrameworkChainSlice
    : chainSliceMod.fetchFrameworkChainSlice;
  const slice = await fetcher({ active_frameworks: activeNames, max_hops: depth });
  // slice is always defined per Plan 02 contract (graceful degradation never returns undefined).
  return slice;
}
```

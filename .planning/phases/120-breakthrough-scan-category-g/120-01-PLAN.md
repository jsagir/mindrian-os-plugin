    ```javascript
          eventType: 'breakthrough_surfaced', limit: 500,
        }) || [];
        const dismissed = navigation.findRecentChanges(roomState.db, since, {
          eventType: 'breakthrough_dismissed', limit: 500,
        }) || [];
        const fired = surfaced.filter(e => e.properties && e.properties.kind === kind).slice(0, D19_FIRE_WINDOW);
        if (fired.length < 10) return false;   // need a minimum sample size before throttling
        const fireIds = new Set(fired.map(e => e.properties && e.properties.breakthrough_id).filter(Boolean));
        const dismissedInWindow = dismissed.filter(e => e.properties &&
          e.properties.kind === kind &&
          fireIds.has(e.properties.breakthrough_id)).length;
        const rate = dismissedInWindow / fired.length;
        return rate > D19_DISMISSAL_THRESHOLD;
      } catch (_e) {
        return false;
      }
    }

    function scoreBreakthrough(candidate, roomState, nowMs) {
      if (!candidate || typeof candidate !== 'object') return 0;
      const conf = (typeof candidate.confidence === 'number') ? candidate.confidence : 0;
      const diff = (typeof candidate.differential === 'number') ? candidate.differential : 0;
      const recency = recencyDecay(candidate.detected_at || (nowMs || Date.now()), nowMs);
      const acLog = artifactCountLog(candidate.artifact_ids);
      const engagement = getUserEngagementPrior(candidate.kind, roomState);
      const W = SCORING_WEIGHTS;
      return (W.confidence * conf)
        + (W.recency * recency)
        + (W.differential * diff)
        + (W.artifact_count_log * acLog)
        + (W.user_engagement_prior * engagement);
    }

    function rankBreakthroughs(candidates, roomState, nowMs) {
      if (!Array.isArray(candidates)) return [];
      const now = nowMs || Date.now();
      const scored = candidates.map(c => ({ candidate: c, score: scoreBreakthrough(c, roomState, now) }));
      // Stable sort: score descending, ties broken by detected_at descending (newer wins).
      scored.sort((a, b) => {
        if (b.score !== a.score) return b.score - a.score;
        return (b.candidate.detected_at || 0) - (a.candidate.detected_at || 0);
      });
      return scored.map(s => ({ ...s.candidate, score: s.score }));
    }

    function pickTopWithAffordance(candidates, roomState, nowMs) {
      const ranked = rankBreakthroughs(candidates, roomState, nowMs);
      if (ranked.length === 0) return { top: null, more_count: 0 };
      return { top: ranked[0], more_count: ranked.length - 1, queued: ranked.slice(1) };
    }

    module.exports = {
      scoreBreakthrough,
      rankBreakthroughs,
      pickTopWithAffordance,
      getUserEngagementPrior,
      isThrottledKind,
      recencyDecay,
      artifactCountLog,
      SCORING_WEIGHTS,
      RECENCY_HALF_LIFE_DAYS,
      ENGAGEMENT_NEUTRAL_PRIOR,
      D19_DISMISSAL_THRESHOLD,
      D19_FIRE_WINDOW,
    };
    ```

    'use strict';
    // Phase 120-02 -- D-13..D-15 resurfacing rules. Per CONTEXT.md verbatim:
    //   D-13 (dismissed): 7-day cooldown MINIMUM, AND only if new artifacts have accumulated.
    //                     BOTH conditions REQUIRED. Time alone does NOT license resurfacing
    //                     (user verbatim 2026-05-16: "that's manipulation").
    //   D-14 (confirmed): once-only. Never resurface. Patronizing.
    //   D-15 (filed-as-decision): never resurface as breakthrough; may be referenced as ENABLES
    //                              in future related breakthroughs via Phase 88 decision-log.
    //
    // Canon Part 8: ALL reads via navigation.cjs chokepoint; no Brain coupling; no cross-room.
    // Canon Part 9: SQL is the local mind; resurfacing is a read-only function over event log.

    const navigation = require('../navigation.cjs');

    const SEVEN_DAY_COOLDOWN_MS = 7 * 24 * 3600 * 1000;

    function findFirstEventForBreakthrough(id, eventType, db) {
      try {
        const since = 0;  // search entire log; resurfacing is once-only signal
        const events = navigation.findRecentChanges(db, since, { eventType: eventType, limit: 500 }) || [];
        return events.find(e => e.properties && e.properties.breakthrough_id === id) || null;
      } catch (_e) {
        return null;
      }
    }

    function hasBeenConfirmed(id, db) {
      return !!findFirstEventForBreakthrough(id, 'breakthrough_confirmed', db);
    }

    function hasBeenFiledAsDecision(id, db) {
      return !!findFirstEventForBreakthrough(id, 'breakthrough_filed_as_decision', db);
    }

    function dismissalCooldownExpired(id, db) {
      const dismiss = findFirstEventForBreakthrough(id, 'breakthrough_dismissed', db);
      if (!dismiss) return true;  // never dismissed -> vacuously eligible re: cooldown
      const ageMs = Date.now() - dismiss.createdAt;
      return ageMs >= SEVEN_DAY_COOLDOWN_MS;
    }

    // Phase 120-02 -- D-19 per-detector dismissal-rate canary. Per CONTEXT.md verbatim:
    //   Track per-detector dismissal rate over a 100-fire rolling window. If rate crosses 30%,
    //   auto-throttle that detector to soft-fire-only until manually reviewed.

    const D19_DISMISSAL_THRESHOLD = 0.30;
    const D19_FIRE_WINDOW = 100;
    const D19_MIN_SAMPLE = 10;

    function computeDismissalRate(kind, db) {
      try {
        if (!kind || !db) return { rate: 0, sample_size: 0, throttled: false };
        const windowMs = 90 * 24 * 3600 * 1000;  // 90-day search window
        const since = Date.now() - windowMs;
        const surfaced = navigation.findRecentChanges(db, since, {
          eventType: 'breakthrough_surfaced', limit: 500,
        }) || [];
        const dismissed = navigation.findRecentChanges(db, since, {
          eventType: 'breakthrough_dismissed', limit: 500,
        }) || [];
        const kindSurfaced = surfaced.filter(e => e.properties && e.properties.kind === kind)
          .slice(0, D19_FIRE_WINDOW);
        const sample = kindSurfaced.length;
        const fireIds = new Set(kindSurfaced.map(e => e.properties.breakthrough_id).filter(Boolean));
        const dismissedInWindow = dismissed.filter(e => e.properties &&
          e.properties.kind === kind &&
          fireIds.has(e.properties.breakthrough_id)).length;
        const rate = sample > 0 ? dismissedInWindow / sample : 0;
        const throttled = sample >= D19_MIN_SAMPLE && rate > D19_DISMISSAL_THRESHOLD;
        return { rate: rate, sample_size: sample, throttled: throttled };
      } catch (_e) {
        return { rate: 0, sample_size: 0, throttled: false };
      }
    }

    function isThrottled(kind, db) {
      return computeDismissalRate(kind, db).throttled;
    }

    function emitThrottleEvent(kind, db, result) {
      const payload = {
        kind: kind,
        rate: result.rate,
        sample_size: result.sample_size,
        threshold: D19_DISMISSAL_THRESHOLD,
        throttled_at: Date.now(),
        source_path: 'system:breakthrough-canary',
        created_by: 'phase-120-canary',
      };
      return navigation.logMemoryEvent(db, 'breakthrough_throttled', payload);
    }

      isThrottled,
      emitThrottleEvent,

        }

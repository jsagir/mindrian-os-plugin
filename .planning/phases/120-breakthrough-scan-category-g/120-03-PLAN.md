      const themePart = theme ? ' on ' + theme : '';
      return "You're seeing a " + kindDisplay + themePart + " " + evidenceCite + " -- " + mechanism + " -- " + timeAnchor + ".";
    }

    function citeEvidence(line) {
      // Rule 1: line MUST contain one of: (artifacts ...), [[...]], (see edges ...)
      if (typeof line !== 'string') return false;
      if (/\(artifacts\s+[^)]+\)/i.test(line)) return true;
      if (/\[\[.+\]\]/.test(line)) return true;
      if (/\(see\s+edges?\s+[^)]+\)/i.test(line)) return true;
      return false;
    }

    function includeMechanism(line) {
      // Rule 2: line MUST contain "by " followed by a verb phrase. Heuristic: "by " + word.
      if (typeof line !== 'string') return false;
      return /\bby\s+\w+/i.test(line);
    }

    function anchorTime(line) {
      // Rule 3: line MUST contain a time-anchor pattern.
      if (typeof line !== 'string') return false;
      const patterns = [
        /in the last\s+\d+\s+(hour|hours|day|days|week|weeks)/i,
        /in the last\s+(hour|day|two days|week|two weeks)/i,
        /this (week|morning|month)/i,
        /today|yesterday|tonight/i,
        /across this week/i,
        /since\s+(monday|tuesday|wednesday|thursday|friday|saturday|sunday|\d{4}-\d{2}-\d{2})/i,
      ];
      return patterns.some(p => p.test(line));
    }

    function noUnbackedSuperlatives(line) {
      // Rule 4: forbidden superlatives MUST NOT appear unless adjacent to numeric backing.
      if (typeof line !== 'string') return true;
      const lower = line.toLowerCase();
      for (const word of FORBIDDEN_SUPERLATIVES) {
        const idx = lower.indexOf(word);
        if (idx >= 0) {
          // Look for numeric backing within 60 chars (either direction).
          const window = lower.slice(Math.max(0, idx - 60), Math.min(lower.length, idx + word.length + 60));
          if (!/\d/.test(window) && !/\(.+\d+.+\)/.test(window)) {
            return false;
          }
        }
      }
      for (const word of FREQUENCY_WORDS) {
        const idx = lower.indexOf(word);
        if (idx >= 0) {
          // Frequency words need an adjacent count.
          const window = lower.slice(Math.max(0, idx - 30), Math.min(lower.length, idx + word.length + 30));
          if (!/\d/.test(window)) return false;
        }
      }
      return true;
    }

    function auditVoiceLine(line) {
      const violations = [];
      if (!citeEvidence(line)) violations.push('evidence_requirement');
      if (!includeMechanism(line)) violations.push('mechanism_clause');
      if (!anchorTime(line)) violations.push('time_anchor');
      if (!noUnbackedSuperlatives(line)) violations.push('no_unbacked_superlatives');
      return { ok: violations.length === 0, violations: violations };
    }

      formatTimeAnchor,

      FORBIDDEN_SUPERLATIVES,
      FREQUENCY_WORDS,
      KIND_DISPLAY,
    };

    'use strict';
    // Phase 120-03 -- SOFT_BAND review queue (.rooms/breakthrough-review-queue.db).
    // Per CONTEXT.md "Claude's Discretion" item 3: a separate SQLite db at the rooms-home level
    // (mirrors Phase 119-01 rooms-meta.db precedent). Sibling pattern; does NOT pollute per-room
    // room.db with cross-room review backlog.
    //
    // Canon Part 8: queue stays LOCAL to the user's machine; no Brain coupling.
    // Canon Part 9: schema is queryable via standard SQLite; sample 20% manually each week.

    const fs = require('node:fs');
    const path = require('node:path');
    const crypto = require('node:crypto');

    let Database;
    try {
      Database = require('better-sqlite3');
    } catch (_e) {
      // Graceful: in environments without better-sqlite3, fall back to a no-op surface.
      Database = null;
    }

    const TABLE_DDL = `CREATE TABLE IF NOT EXISTS review_candidates (
      id TEXT PRIMARY KEY,
      room_slug TEXT,
      breakthrough_id TEXT NOT NULL,
      kind TEXT NOT NULL,
      confidence REAL NOT NULL,
      theme TEXT,
      artifact_ids_json TEXT,
      queued_at INTEGER NOT NULL,
      reviewed_at INTEGER,
      review_status TEXT NOT NULL DEFAULT 'pending'
    )`;

    function openReviewQueue(roomsHome) {
      if (!Database) {
        return { db: null, fallback: true, reason: 'better_sqlite3_not_available' };
      }
      try {
        const metaDir = path.join(roomsHome, '.rooms');
        fs.mkdirSync(metaDir, { recursive: true, mode: 0o755 });
        const dbPath = path.join(metaDir, 'breakthrough-review-queue.db');
        const db = new Database(dbPath);
        db.exec(TABLE_DDL);
        return { db: db, dbPath: dbPath, fallback: false };
      } catch (err) {
        // Graceful: fall back to in-memory db so the scanner can still insert; not persistent.
        try {
          const db = new Database(':memory:');
          db.exec(TABLE_DDL);
          return { db: db, fallback: true, reason: 'rooms_home_not_writable:' + err.message.slice(0, 80) };
        } catch (err2) {
          return { db: null, fallback: true, reason: 'open_failed:' + err2.message.slice(0, 80) };
        }
      }
    }

    function insertReviewCandidate(db, breakthrough, roomSlug) {
      if (!db) return { ok: false, reason: 'no_db' };
      const bk = breakthrough || {};
      const queueId = 'review:' + crypto.randomBytes(8).toString('hex');
      const queued_at = Date.now();
      try {
        db.prepare(
          "INSERT INTO review_candidates (id, room_slug, breakthrough_id, kind, confidence, theme, artifact_ids_json, queued_at, review_status) " +
          "VALUES (?, ?, ?, ?, ?, ?, ?, ?, 'pending')"
        ).run(
          queueId,
          (typeof roomSlug === 'string' ? roomSlug : null),
          bk.id || null,
          bk.kind || null,
          (typeof bk.confidence === 'number' ? bk.confidence : 0),
          (typeof bk.theme === 'string' ? bk.theme.slice(0, 200) : null),
          JSON.stringify(Array.isArray(bk.artifact_ids) ? bk.artifact_ids : []),
          queued_at
        );
        return { ok: true, queue_id: queueId, queued_at: queued_at };
      } catch (err) {
        return { ok: false, reason: err.message };
      }
    }

    function listPendingReviews(db, limit) {
      if (!db) return [];
      const lim = (Number.isInteger(limit) && limit > 0) ? limit : 100;
      try {
        return db.prepare(
          "SELECT id, room_slug, breakthrough_id, kind, confidence, theme, queued_at FROM review_candidates WHERE review_status = 'pending' ORDER BY queued_at DESC LIMIT ?"
        ).all(lim);
      } catch (_e) {
        return [];
      }
    }

    const REVIEW_QUEUE_DB_PATH = '.rooms/breakthrough-review-queue.db';

    module.exports = {
      openReviewQueue,
      insertReviewCandidate,
      listPendingReviews,
      REVIEW_QUEUE_DB_PATH,
      TABLE_DDL,
    };

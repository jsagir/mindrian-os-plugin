```javascript
async function createOpportunityNode(conn, opp) {
  const esc = s => String(s || '').replace(/'/g, "\\'");
  await conn.query(`
    MERGE (o:Opportunity {problem_hash: '${esc(opp.problem_hash)}'})
    SET o.problem = '${esc(opp.problem)}',
        o.mirror_solution = '${esc(opp.mirror_solution)}',
        o.domain = '${esc(opp.domain)}',
        o.knight_position = '${esc(opp.knight_position)}',
        o.confidence = ${parseFloat(opp.confidence) || 0},
        o.source_framework = '${esc(opp.source_framework)}',
        o.status = '${esc(opp.status || 'banked')}',
        o.created = '${esc(new Date().toISOString())}'
  `);
  return { problem_hash: opp.problem_hash };
}
```

```javascript
async function indexOpportunity(roomDir, opportunity) {
  return enqueueWrite(roomDir, async () => {
    const { db, conn } = await lazygraph.openGraph(roomDir);
    try {
      // Create opportunity node (MERGE = idempotent)
      await lazygraph.createOpportunityNode(conn, opportunity);

      // ADDRESSES edge: link to artifacts in the source section
      // Find artifacts in the opportunity's domain section
      if (opportunity.domain) {
        const artResult = await conn.query(
          `MATCH (a:Artifact {section: '${String(opportunity.domain || '').replace(/'/g, "\\'")}'}) RETURN a.id AS id LIMIT 5`
        );
        const rows = await lazygraph.queryGraph(conn,
          `MATCH (a:Artifact {section: '${String(opportunity.domain || '').replace(/'/g, "\\'")}'}) RETURN a.id AS id LIMIT 5`
        );
        for (const row of rows) {
          const aid = row.id || row[0];
          if (aid) await lazygraph.createAddressesEdge(conn, opportunity.problem_hash, aid);
        }
      }

      // IN_DOMAIN edge: link to section node
      if (opportunity.domain) {
        await lazygraph.createInDomainEdge(conn, opportunity.problem_hash, opportunity.domain);
      }

      return { success: true, problem_hash: opportunity.problem_hash };
    } finally {
      await lazygraph.closeGraph(db);
    }
  });
}
```

IMPORTANT: Use the exact `enqueueWrite` pattern from `persistDecisionEdge` -- this ensures serialized KuzuDB writes. Do NOT open a second connection or bypass the write queue.

```javascript
function filterOpportunities(roomDir, filters = {}) {
  const { opportunities } = listOpportunities(roomDir);
  let filtered = opportunities;

  // Read full frontmatter for filtering (listOpportunities only returns grant fields)
  const oppDir = path.join(path.resolve(roomDir), 'opportunity-bank');
  const enriched = filtered.map(opp => {
    try {
      const content = fs.readFileSync(path.join(oppDir, opp.filename), 'utf8');
      const fm = parseOpportunityFrontmatter(content);
      return { ...opp, ...fm };
    } catch (_e) {
      return opp;
    }
  });

  let result = enriched;

  if (filters.domain) {
    result = result.filter(o => o.domain && o.domain.toLowerCase().includes(filters.domain.toLowerCase()));
  }

  if (filters.knight) {
    result = result.filter(o => o.knight_position === filters.knight);
  }

  if (filters.minConfidence != null) {
    const min = parseFloat(filters.minConfidence);
    result = result.filter(o => o.confidence != null && parseFloat(o.confidence) >= min);
  }

  return { opportunities: result, count: result.length, filtered_from: enriched.length };
}
```

graph-links.cjs Cypher -> SQL conversions:
```
getPageLinks:
  CYPHER: MATCH (a:Artifact {id: '$id'})-[r]->(b:Artifact) RETURN type(r) AS rel, b.id AS target, b.title AS title
  SQL:    SELECT e.type AS rel, e.target, json_extract(n.properties, '$.title') AS title
          FROM edges e JOIN nodes n ON e.target = n.id
          WHERE e.source = ? AND n.type = 'Artifact'

getBacklinks:
  CYPHER: MATCH (a:Artifact)-[r]->(b:Artifact {id: '$id'}) RETURN type(r) AS rel, a.id AS source, a.title AS title
  SQL:    SELECT e.type AS rel, e.source, json_extract(n.properties, '$.title') AS title
          FROM edges e JOIN nodes n ON e.source = n.id
          WHERE e.target = ? AND n.type = 'Artifact'

getSeeAlso (CONVERGES):
  CYPHER: MATCH (a:Artifact {id: '$id'})-[r:CONVERGES]->(b:Artifact) RETURN b.id, b.title, r.term
  SQL:    SELECT e.target, json_extract(n.properties, '$.title') AS title, json_extract(e.properties, '$.term') AS theme
          FROM edges e JOIN nodes n ON e.target = n.id
          WHERE e.source = ? AND e.type = 'CONVERGES'

getSeeAlso (ENABLES):
  SQL:    SELECT e.target, json_extract(n.properties, '$.title') AS title
          FROM edges e JOIN nodes n ON e.target = n.id
          WHERE e.source = ? AND e.type = 'ENABLES'

getGraphData (nodes):
  SQL:    SELECT id, json_extract(properties, '$.title') AS title, json_extract(properties, '$.section') AS section
          FROM nodes WHERE type = 'Artifact'

getGraphData (edges per type):
  SQL:    SELECT source, target FROM edges WHERE type = ?
```

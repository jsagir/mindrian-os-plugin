// Database: room/.mindrian/room.db

//         edges (source TEXT, target TEXT, type TEXT, properties TEXT JSON, PK(source,target,type))

// Query all nodes of a type:
"SELECT id, type, properties FROM nodes WHERE type = ?"
// Access JSON properties:
"SELECT id, json_extract(properties, '$.title') AS title, json_extract(properties, '$.section') AS section FROM nodes WHERE type = 'Artifact'"
// Query edges:
"SELECT source, target, type, properties FROM edges WHERE type = ?"
// Query edges with node joins:
"SELECT e.source, e.target, e.type, e.properties, json_extract(n1.properties, '$.title') AS src_title FROM edges e JOIN nodes n1 ON e.source = n1.id WHERE e.type = ?"

Cypher-to-SQL mapping for build-graph-from-kuzu.cjs:
- `MATCH (s:Section) RETURN s.name, s.label` -> `SELECT id, json_extract(properties, '$.name') AS name, json_extract(properties, '$.label') AS label FROM nodes WHERE type = 'Section'`
- `MATCH (a:Artifact) RETURN a.id, a.title, a.section, a.methodology, a.created` -> `SELECT id, json_extract(properties, '$.title') AS title, json_extract(properties, '$.section') AS section, json_extract(properties, '$.methodology') AS methodology, json_extract(properties, '$.created') AS created FROM nodes WHERE type = 'Artifact'`
- `MATCH (a)-[r]->(b) RETURN coalesce(a.id,a.name), label(r), coalesce(b.id,b.name)` -> `SELECT e.source AS src, e.type AS relType, e.target AS tgt, e.properties FROM edges e`
- Type-specific edge queries: `SELECT source, target, properties FROM edges WHERE type = 'HSI_CONNECTION'`

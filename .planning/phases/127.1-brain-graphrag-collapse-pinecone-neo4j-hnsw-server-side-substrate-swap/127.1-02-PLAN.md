    let driver = null;
    let neo4jVersionCache = null;  // memoized at first call to pick SEARCH vs queryNodes()

    function getDriver() {
      if (!driver) {
        driver = neo4j.driver(
          process.env.NEO4J_URI,

      return driver;

    async function detectNeo4jVersion(session) {
      if (neo4jVersionCache !== null) return neo4jVersionCache;
      const result = await session.run('CALL dbms.components() YIELD versions RETURN versions[0] AS version');
      const v = result.records[0].get('version');

      return v;

    // Returns true if SEARCH clause is available (Neo4j >= 2026.04 per RESEARCH A127.1.1).
    function supportsSearchClause(version) {
      const m = /^(\d+)\.(\d+)/.exec(version);
      if (!m) return false;
      const major = parseInt(m[1], 10);
      // Neo4j calendar versioning: 2026.04 maps to major=2026.
      return major >= 2026;
    }

    async function searchSemantic({ embedding, topK = 5, namespace = null, session = null }) {
      if (!Array.isArray(embedding) || embedding.length !== EMBEDDING_DIMS) {
        throw new Error('embedding must be a 1024-dim Float64 array; observed length=' + (embedding && embedding.length));
      }
      const ownSession = !session;
      const sess = session || getDriver().session({ defaultAccessMode: neo4j.session.READ });
      try {
        const version = await detectNeo4jVersion(sess);
        const useSearch = supportsSearchClause(version);

        // queryNodes() path (works on Neo4j 5.13 through 2026.03; deprecated 2026.04+).
        const cypher = useSearch
          ? `CALL db.index.vector.queryNodes($indexName, $topK, $queryVec)
             YIELD node, score
             WITH node, score
             ${namespace ? "WHERE node.namespace = $namespace" : ""}
             RETURN node.id AS id,
                    node.metadata AS metadata_json,
                    score
             ORDER BY score DESC
             LIMIT $topK`
          : `CALL db.index.vector.queryNodes($indexName, $topK, $queryVec)
             YIELD node, score
             ${namespace ? "WHERE node.namespace = $namespace" : ""}
             RETURN node.id AS id,
                    node.metadata AS metadata_json,
                    score
             ORDER BY score DESC
             LIMIT $topK`;

        const params = { indexName: INDEX_NAME, topK: neo4j.int(topK), queryVec: embedding };
        if (namespace) params.namespace = namespace;

        const result = await sess.run(cypher, params);
        const hits = result.records.map(r => {
          const metadataJson = r.get('metadata_json');
          let metadata = {};
          try { metadata = metadataJson ? JSON.parse(metadataJson) : {}; } catch (_) { metadata = {}; }
          return {
            score: r.get('score'),
            id: r.get('id'),
            text: metadata.text || metadata.chunk_text || '',
            metadata: metadata
          };
        });
        return { hits, engine: 'neo4j-hnsw', version, used_search_clause: useSearch };
      } finally {
        if (ownSession) await sess.close();
      }
    }

    function registerVectorSearchTool(server) {
      server.tool(
        'brain_search',
        'Semantic search across Brain knowledge (1427 PWS methodology vectors). Backed by Neo4j HNSW (Phase 127.1 cutover). Namespaces: core (default). Returns top-K hits ranked by cosine similarity.',
        {
          query_vector: z.array(z.number()).length(EMBEDDING_DIMS).describe('Pre-computed 1024-dim multilingual-e5-large embedding'),
          namespace: z.string().optional().describe('Namespace filter (default: all)'),
          topK: z.number().optional().describe('Number of results (default 5)')
        },
        async ({ query_vector, namespace, topK }) => {
          try {
            const out = await searchSemantic({ embedding: query_vector, topK: topK || 5, namespace: namespace || null });
            return { content: [{ type: 'text', text: JSON.stringify(out, null, 2) }] };
          } catch (err) {
            return { content: [{ type: 'text', text: 'Error: ' + err.message }], isError: true };
          }
        }
      );
    }

    module.exports = { registerVectorSearchTool, searchSemantic, INDEX_NAME, EMBEDDING_DIMS, SIMILARITY_METRIC };

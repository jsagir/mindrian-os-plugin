```javascript
async function embedViaPineconeInference(text) {
  const pc = getPinecone();
  const resp = await pc.inference.embed({
    model: 'multilingual-e5-large',
    inputs: [{ text }],
    parameters: { input_type: 'query' }
  });
  return resp.data[0].values;  // 1024-float array
}
```

`scripts/sessionstart-npm-reconcile.cjs` shape (defensive, CJS, node builtins only):
```js
#!/usr/bin/env node
'use strict';
// Phase 95.6 D-05d -- SessionStart runtime npm-dependency reconciliation.
// Idempotent: only runs `npm install` if node_modules is missing or out of
// sync with package.json dependencies. Defensive: ANY error -> {continue:true}.
// Canon Part 8: zero network surface beyond `npm install` itself; no Brain,
// no Tavily, no user data. Pure CJS, node builtins only.
process.on('uncaughtException', () => { try { process.stdout.write(JSON.stringify({ continue: true })); } catch (e) {} process.exit(0); });
const fs = require('node:fs');
const path = require('node:path');
const PLUGIN_ROOT = process.env.CLAUDE_PLUGIN_ROOT || process.env.MINDRIAN_OS_ROOT
  || path.join(require('node:os').homedir(), '.claude', 'plugins', 'mindrian-os');
let needInstall = false;
try {
  const pkgPath = path.join(PLUGIN_ROOT, 'package.json');
  const pkg = JSON.parse(fs.readFileSync(pkgPath, 'utf8'));
  const deps = Object.keys(pkg.dependencies || {});
  const nm = path.join(PLUGIN_ROOT, 'node_modules');
  if (!fs.existsSync(nm)) { needInstall = deps.length > 0; }
  else {
    for (const d of deps) { if (!fs.existsSync(path.join(nm, ...d.split('/')))) { needInstall = true; break; } }
  }
  if (needInstall) {
    const { spawnSync } = require('node:child_process');
    spawnSync('npm', ['install', '--no-audit', '--no-fund', '--silent'], { cwd: PLUGIN_ROOT, timeout: 60000 });
  }
} catch (e) { /* swallow */ }
process.stdout.write(JSON.stringify({ continue: true }));
```
Keep it lean. No emoji, no em-dashes. The `timeout: 60000` bounds the npm call. The `--silent` keeps the hook quiet on a no-op session start.

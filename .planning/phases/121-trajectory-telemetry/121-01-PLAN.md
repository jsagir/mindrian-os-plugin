    const fs = require('node:fs');
    const path = require('node:path');
    const os = require('node:os');
    const crypto = require('node:crypto');

    const REPO = path.resolve(__dirname, '..');
    const writer = require(path.join(REPO, 'lib/core/telemetry/writer.cjs'));

    const { ALLOWED_FIELDS } = require(path.join(REPO, 'lib/core/telemetry/schema.cjs'));

    function homeDir() {
      return process.env.HOME || process.env.USERPROFILE || os.homedir();
    }

    function telemetryRoot() {
      return path.join(homeDir(), '.mindrian', 'telemetry');
    }

    // 4 migration sources with their shapes + normalization mappers.

      {
        name: 'mva',

        normalize: (row) => row  // event field already matches EVENT_TYPES (mva_*)

    function normalizeSelectorRow(row) {

      // For Plan 121-01 we coerce to command_invocation (Plan 121-02 wires the proper
      // selector_pick capture point with sub_shape/mode/ranker_confidence). Legacy rows

      const ratio = Number(row.ratio);

        outcome: 'served',

        context_hash: sha256(String(row.room_slug || '')).slice(0, 16)
      };
    }

        timestamp: row.timestamp || new Date(0).toISOString(),
        op: String(row.op || '').slice(0, 64),

        room_slug_sha256: String(row.room_slug_sha256 || sha256(String(row.room || ''))).slice(0, 64)
      };
    }

    function normalizeQueryEfficiencyRow(row) {
      return normalizeSelectorRow(row); // same shape
    }

    function sha256(s) {
      return crypto.createHash('sha256').update(s).digest('hex');
    }

    function isoWeekKey(date) {
      // mirrors writer.cjs isoWeekFilename minus the prefix/suffix
      const d = new Date(Date.UTC(date.getUTCFullYear(), date.getUTCMonth(), date.getUTCDate()));
      const dayNum = d.getUTCDay() || 7;
      d.setUTCDate(d.getUTCDate() + 4 - dayNum);
      const yearStart = new Date(Date.UTC(d.getUTCFullYear(), 0, 1));
      const weekNum = Math.ceil(((d - yearStart) / 86400000 + 1) / 7);
      return `${d.getUTCFullYear()}-W${String(weekNum).padStart(2, '0')}`;
    }

    // Idempotence: sha256 of first 5 timestamps in each source.
    function sourceFingerprint(filePath) {
      if (!fs.existsSync(filePath)) return null;
      const text = fs.readFileSync(filePath, 'utf8');
      const ts = [];
      for (const line of text.split('\n')) {
        if (!line) continue;
        try {
          const row = JSON.parse(line);
          const t = row.timestamp || row.ts;
          if (t) ts.push(t);
          if (ts.length >= 5) break;
        } catch (_) { continue; }
      }
      return ts.length > 0 ? sha256(ts.join('|')) : null;
    }

    function alreadyMerged(fingerprint) {
      // Walk all events-*.jsonl looking for a marker line we wrote on first migration.
      const v113Dir = path.join(telemetryRoot(), 'v1.13');
      if (!fs.existsSync(v113Dir)) return false;
      const markerPath = path.join(v113Dir, '.migration-fingerprints.json');
      if (!fs.existsSync(markerPath)) return false;
      try {
        const seen = JSON.parse(fs.readFileSync(markerPath, 'utf8'));
        return Array.isArray(seen.fingerprints) && seen.fingerprints.includes(fingerprint);
      } catch (_) { return false; }
    }

    function recordFingerprint(fingerprint) {
      const v113Dir = path.join(telemetryRoot(), 'v1.13');
      fs.mkdirSync(v113Dir, { recursive: true });
      const markerPath = path.join(v113Dir, '.migration-fingerprints.json');
      let prior = { fingerprints: [] };
      if (fs.existsSync(markerPath)) {
        try { prior = JSON.parse(fs.readFileSync(markerPath, 'utf8')); } catch (_) {}
      }
      if (!prior.fingerprints.includes(fingerprint)) prior.fingerprints.push(fingerprint);
      fs.writeFileSync(markerPath, JSON.stringify(prior, null, 2));
    }

    function migrateOne(source) {
      const p = source.path();
      const result = { name: source.name, migrated: 0, quarantined: 0, idempotent_skipped: 0, renamed: false };
      if (!fs.existsSync(p)) return result;

      const fingerprint = sourceFingerprint(p);
      if (fingerprint && alreadyMerged(fingerprint)) {
        result.idempotent_skipped = 1;
        return result;
      }

      const text = fs.readFileSync(p, 'utf8');
      const rows = [];
      for (const line of text.split('\n')) {
        if (!line) continue;
        try { rows.push(JSON.parse(line)); } catch (_) { continue; }
      }

      // Sort by timestamp ASC for stable replay order.
      rows.sort((a, b) => {
        const ta = a.timestamp || a.ts || '';
        const tb = b.timestamp || b.ts || '';
        return ta.localeCompare(tb);
      });

      for (const raw of rows) {
        let normalized;
        try { normalized = source.normalize(raw); } catch (_) {
          result.quarantined++;
          continue;
        }
        // Strip top-level fields not in the destination event payload schema.
        const eventType = normalized.event;
        const allowed = ALLOWED_FIELDS[eventType];
        if (!allowed) { result.quarantined++; continue; }
        const payload = {};
        for (const k of allowed) {
          if (normalized[k] !== undefined) payload[k] = normalized[k];
        }
        const v = validateEventPayload(eventType, payload);
        if (!v.ok) {
          result.quarantined++;
          appendQuarantine(source.name, raw, v.error);
          continue;
        }
        // Bypass writer.emit() to preserve historical timestamp instead of "now".
        // Direct append to the correct events-<week>.jsonl file.
        appendUnified(eventType, payload, normalized.timestamp);
        result.migrated++;
      }

      // Rename source after successful merge.
      if (fingerprint) recordFingerprint(fingerprint);
      const backupPath = p + '.pre-v121.bak';
      fs.renameSync(p, backupPath);
      result.renamed = true;
      return result;
    }

    function appendUnified(event, payload, historicalTimestamp) {
      const ts = historicalTimestamp || new Date().toISOString();
      const date = new Date(ts);
      const v113Dir = path.join(telemetryRoot(), 'v1.13');
      fs.mkdirSync(v113Dir, { recursive: true });
      const fname = `events-${isoWeekKey(date)}.jsonl`;
      const target = path.join(v113Dir, fname);
      const record = Object.assign(
        { event, schema_version: 1, timestamp: ts, session_id: 'migrated' },
        payload
      );
      fs.appendFileSync(target, JSON.stringify(record) + '\n', 'utf8');
    }

    function appendQuarantine(sourceName, originalRow, reason) {
      const v113Dir = path.join(telemetryRoot(), 'v1.13');
      fs.mkdirSync(v113Dir, { recursive: true });
      const qPath = path.join(v113Dir, `.quarantine-${sourceName}.jsonl`);
      const line = JSON.stringify({
        source: sourceName, reason: String(reason).slice(0, 60), original_sha256: sha256(JSON.stringify(originalRow))
      }) + '\n';
      fs.appendFileSync(qPath, line, 'utf8');
    }

    function main() {
      const summary = {
        migrated: 0, quarantined: 0, idempotent_skipped: 0,
        sources_renamed: [], per_source: []
      };
      for (const s of SOURCES) {
        const r = migrateOne(s);
        summary.migrated += r.migrated;
        summary.quarantined += r.quarantined;
        summary.idempotent_skipped += r.idempotent_skipped;
        if (r.renamed) summary.sources_renamed.push(s.name);
        summary.per_source.push(r);
      }
      process.stdout.write(JSON.stringify(summary, null, 2) + '\n');
      process.exit(0);
    }

    if (require.main === module) main();

    module.exports = {
      migrateOne, sourceFingerprint, isoWeekKey, sha256,
      normalizeSelectorRow, normalizeNavBypassRow, normalizeQueryEfficiencyRow,
      SOURCES
    };

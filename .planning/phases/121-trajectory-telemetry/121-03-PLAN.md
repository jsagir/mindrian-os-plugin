    #!/usr/bin/env node
    'use strict';

    const path = require('node:path');
    const REPO = path.resolve(__dirname, '..');
    const telemetry = require(path.join(REPO, 'lib/core/telemetry/writer'));

    function parseFlag(args, name) {
      const m = args.find(a => a.startsWith(`--${name}=`));
      if (!m) return undefined;
      return m.slice(`--${name}=`.length);
    }

    function parseBool(v, field) {
      if (v === undefined) throw new Error(`missing flag: --${field}`);
      const lower = String(v).toLowerCase();
      if (['true', 'yes', '1'].includes(lower)) return true;
      if (['false', 'no', '0'].includes(lower)) return false;
      throw new Error(`invalid boolean for --${field}: ${v}`);
    }

    function main(argv) {
      try {
        const engagedPast15m = parseBool(parseFlag(argv, 'engaged-past-15m'), 'engaged-past-15m');
        const handedBackMaterial = parseBool(parseFlag(argv, 'handed-back-material'), 'handed-back-material');
        const returnedWithin48h = parseBool(parseFlag(argv, 'returned-within-48h'), 'returned-within-48h');
        const ttrRaw = parseFlag(argv, 'ttr-seconds');
        if (ttrRaw === undefined) throw new Error('missing flag: --ttr-seconds');
        const ttrSeconds = parseInt(ttrRaw, 10);
        if (!Number.isFinite(ttrSeconds) || ttrSeconds < 0) throw new Error('ttr-seconds must be a non-negative integer');
        const testerIdHash = parseFlag(argv, 'tester-id-hash');
        if (!testerIdHash || !/^[0-9a-f]{64}$/.test(testerIdHash)) {
          throw new Error('tester-id-hash must be a 64-char lowercase hex sha256');
        }

        telemetry.emit('empathy_observation', {
          engaged_past_15m: engagedPast15m,
          handed_back_material: handedBackMaterial,
          returned_within_48h: returnedWithin48h,
          ttr_seconds: ttrSeconds,
          tester_id_hash: testerIdHash
        });
        process.exit(0);
      } catch (e) {
        process.stderr.write(`empathy-observation-emit: ${e.message}\n`);
        process.exit(1);
      }
    }

    if (require.main === module) main(process.argv.slice(2));

    module.exports = { main };

     * Copyright (c) 2026 Mindrian. BSL 1.1.

     * Phase 121-03 D-09: emit helper for room_receipt_written events.

     *

    'use strict';

    const crypto = require('node:crypto');
    const telemetry = require('./telemetry/writer');

    function _sha256(s) {
      return crypto.createHash('sha256').update(String(s || '')).digest('hex');
    }

    /**
     * emitReceiptWritten(roomSlug, conversationId) -> void

     * Non-throwing. Best-effort. Validation breaches are swallowed so the

        });
      } catch (_e) { /* swallow */ }
    }

    module.exports = { emitReceiptWritten };

    #!/usr/bin/env node
    'use strict';

    const path = require('node:path');
    const crypto = require('node:crypto');
    const REPO = path.resolve(__dirname, '..');
    const telemetry = require(path.join(REPO, 'lib/core/telemetry/writer'));

    function sha256(s) {
      return crypto.createHash('sha256').update(String(s || '')).digest('hex');
    }

    function main() {
      try {
        // Hook env contract (set by Claude Code PostToolUse):
        //   CLAUDE_TOOL_NAME      -- 'Bash' / 'Edit' / 'Read' / etc.
        //   CLAUDE_TOOL_COMMAND   -- for SlashCommand: '/mos:scout'
        //   CLAUDE_TOOL_OUTCOME   -- 'success' | 'error' | 'cancelled'
        //   CLAUDE_TOOL_DURATION_MS -- integer ms
        //   CLAUDE_SESSION_ID     -- session id
        //   PWD                   -- cwd

        const command = String(process.env.CLAUDE_TOOL_COMMAND || '').slice(0, 64);
        // Filter: only /mos:* invocations.
        if (!/^\/mos:/.test(command)) {
          process.exit(0);
        }

        const outcomeRaw = String(process.env.CLAUDE_TOOL_OUTCOME || 'success').slice(0, 64);
        const outcome = ['success', 'error', 'cancelled'].includes(outcomeRaw) ? outcomeRaw : 'success';
        const durationMs = parseInt(process.env.CLAUDE_TOOL_DURATION_MS || '0', 10) || 0;
        const contextSource = String(process.env.PWD || '') + '|' + String(process.env.CLAUDE_SESSION_ID || '');
        const contextHash = sha256(contextSource).slice(0, 16);

        telemetry.emit('command_invocation', {
          command,
          outcome,
          duration_ms: durationMs,
          context_hash: contextHash
        });
        process.exit(0);
      } catch (_e) {
        // Best-effort: PostToolUse hook MUST NOT crash the user's flow.
        process.exit(0);
      }
    }

    if (require.main === module) main();
    module.exports = { main };

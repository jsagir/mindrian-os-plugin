  function readRoomConfigPrivacyMode(roomDir) {
    if (!roomDir) return null;
    try { const p = path.join(roomDir, '.config.json'); if (!fs.existsSync(p)) return null;
      const cfg = JSON.parse(fs.readFileSync(p, 'utf8'));
      const v = cfg && cfg.preferences && cfg.preferences.brain_privacy_mode;
      return MODES.includes(v) ? v : null; } catch (_) { return null; }
  }
  function roomHasExcerptApproval(db, roomDir) {
    // The Part-3 Decision Gate writes an APPROVE-with-reason edge / decision node tagged `brain_excerpts` on the room node
    // when the user picks APPROVE on the "excerpts of your room content will reach the Brain" F.0/F.1 selector
    // (rendered via lib/hmi/shape-f0-renderer.cjs). Check the local graph for it via navigation.cjs (or a direct
    // SELECT against decision/edge rows if navigation.cjs is not in scope here -- packet.cjs already takes `db`).
    // As of v1.13.0-beta.3 there is NO shipped consumer of allow_excerpts -- so this returns false unless a
    // brain_excerpts-tagged APPROVE row exists. Implement it as a single SELECT for a node of type 'decision' (or an
    // edge) whose properties JSON includes "brain_excerpts" AND review_status='confirmed' AND created_by='user'.
    try { if (!db) return false;
      const row = db.prepare("SELECT 1 FROM nodes WHERE type IN ('decision','memory_event') AND review_status='confirmed' AND created_by='user' AND instr(properties, 'brain_excerpts') > 0 LIMIT 1").get();
      return !!row; } catch (_) { return false; }
  }
  function resolvePrivacyMode(db, roomDir, opts) {
    const perCall = opts && opts.privacyMode;
    const fromConfig = readRoomConfigPrivacyMode(roomDir);
    const requested = (perCall && MODES.includes(perCall)) ? perCall : (fromConfig || 'local_summary_only');
    if (requested === 'allow_excerpts' && !roomHasExcerptApproval(db, roomDir)) {
      return fromConfig === 'allow_filenames' ? 'allow_filenames' : 'local_summary_only';
    }
    return requested;
  }

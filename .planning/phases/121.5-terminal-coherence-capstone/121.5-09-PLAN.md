    ```
    #!/usr/bin/env node
    'use strict';
    const fs = require('fs');
    const path = require('path');
    const os = require('os');
    const { execSync, spawnSync } = require('child_process');

    const REPO = path.join(__dirname, '..');

    function step1_cold_start() {
      // Invoke SessionStart Coordinator with synthetic worst-case
      const result = spawnSync('node', [path.join(REPO, 'scripts', 'sessionstart-coordinator.cjs')], {
        encoding: 'utf8',
        env: { ...process.env, MINDRIAN_SMOKE_TEST: '1' },
      });
      const envelope = JSON.parse(result.stdout || '{}');
      const body = (envelope.hookSpecificOutput && envelope.hookSpecificOutput.additionalContext) || '';
      const banner = execSync('bash ' + path.join(REPO, 'scripts', 'banner'), { encoding: 'utf8' });
      return {
        envelope_length: body.length,
        envelope_under_budget: body.length <= 2000,
        envelope_has_version: /MindrianOS v\d/.test(body),
        envelope_mentions_larry: /Larry/i.test(body),
        banner_has_version: /MindrianOS v\d/.test(banner),
      };
    }

    function step2_first_material_upload() {
      const tmpRoom = fs.mkdtempSync(path.join(os.tmpdir(), 'coherence-room-'));
      fs.writeFileSync(path.join(tmpRoom, 'STATE.md'), '---\nstage: Pre-Opportunity\n---\n\n# Fresh room\n');
      const sectionDir = path.join(tmpRoom, 'problem-definition');
      fs.mkdirSync(sectionDir, { recursive: true });
      fs.writeFileSync(path.join(sectionDir, 'first-artifact.md'), '# First artifact\n\nThe problem...\n');
      // Run coordinator pointed at this room (the room-aware contributors read STATE.md)
      const result = spawnSync('node', [path.join(REPO, 'scripts', 'sessionstart-coordinator.cjs')], {
        encoding: 'utf8',
        cwd: tmpRoom,
        env: { ...process.env, MINDRIAN_ROOM_DIR: tmpRoom },
      });
      const envelope = JSON.parse(result.stdout || '{}');
      const body = (envelope.hookSpecificOutput && envelope.hookSpecificOutput.additionalContext) || '';
      // Coherence assertion: NOT a flood -- specifically, the body has <= one "Larry:" attribution line, not N
      const larryAttrCount = (body.match(/^\s*Larry:/gm) || []).length;
      return {
        body_length: body.length,
        single_coherent_voice: larryAttrCount <= 1,
        tmp_room: tmpRoom,
      };
    }

    function step3_mos_command() {
      const modes = [];
      for (const env of [
        { COLORTERM: 'truecolor', TERM: 'xterm-256color' },
        { COLORTERM: '', TERM: 'xterm-256color' },
        { CLAUDE_DESKTOP: '1' },
      ]) {
        const result = spawnSync('node', [path.join(REPO, 'scripts', 'help-renderer.cjs')], {
          encoding: 'utf8',
          env: { ...process.env, ...env },
        });
        modes.push({
          env_summary: JSON.stringify(env),
          has_groups: ['Getting Started', 'Problem Discovery', 'Working', 'Export', 'Publish', 'Hub'].every(g => result.stdout.includes(g)),
          has_ansi_codes: /\x1b\[/.test(result.stdout),
          length: result.stdout.length,
        });
      }
      return { modes, all_groups_present: modes.every(m => m.has_groups) };
    }

    function step4_decision_gate() {
      // Cannot interactively invoke AskUserQuestion. Instead: assert SKILL.md drift check passes + assert F-frame fixture renders.
      const driftResult = spawnSync('node', [path.join(REPO, 'scripts', 'check-skill-vs-code-drift.cjs')], { encoding: 'utf8' });
      // Synthetic F.1 frame fixture (the rendering of an F.1 selector header per SKILL.md §2):
      const fFrame = ['■ [CONTEXT] - NEXT MOVE             - decision gate',
                      '▼ LOCAL   / BRAIN   / SIGNAL',
                      '▶ Choose next move:',
                      '  1. Run Methodology -- /mos:beautiful-question',
                      '  2. Defer            -- look around first',
                      '  3. Free-Text        -- tell Larry what you want'].join('\n');
      return {
        skill_drift_exit_code: driftResult.status,
        skill_drift_valid: driftResult.status === 0,
        f_frame_has_filled_square: fFrame.startsWith('■'),
        f_frame_has_tri_context: fFrame.includes('LOCAL') && fFrame.includes('BRAIN') && fFrame.includes('SIGNAL'),
        f_frame_has_free_text_last: /Free-Text/.test(fFrame.split('\n').pop()),
      };
    }

    function step5_tension_resurface() {
      // Stage a synthetic tension in a temp ~/.mindrian
      const tmpHome = fs.mkdtempSync(path.join(os.tmpdir(), 'coherence-home-'));
      const tensionDir = path.join(tmpHome, '.mindrian', 'tensions');
      fs.mkdirSync(tensionDir, { recursive: true });
      fs.writeFileSync(path.join(tensionDir, 'pending.json'), JSON.stringify({ count: 1, last_seen: new Date().toISOString() }));
      const result = spawnSync('node', [path.join(REPO, 'scripts', 'sessionstart-coordinator.cjs')], {
        encoding: 'utf8',
        env: { ...process.env, HOME: tmpHome, MINDRIAN_SYNTHETIC_WORST_CASE: '1' },
      });
      const envelope = JSON.parse(result.stdout || '{}');
      const body = (envelope.hookSpecificOutput && envelope.hookSpecificOutput.additionalContext) || '';
      return {
        envelope_length: body.length,
        envelope_under_budget: body.length <= 2000,
        tension_present: /tension|unresolved/i.test(body),
      };
    }

    function tri_polar_checks() {
      const cli = spawnSync('node', [path.join(REPO, 'scripts', 'help-renderer.cjs')], { encoding: 'utf8', env: { ...process.env, COLORTERM: 'truecolor' } });
      const desktop = spawnSync('node', [path.join(REPO, 'scripts', 'help-renderer.cjs')], { encoding: 'utf8', env: { ...process.env, CLAUDE_DESKTOP: '1' } });
      const cowork = spawnSync('node', [path.join(REPO, 'scripts', 'help-renderer.cjs')], { encoding: 'utf8', env: { ...process.env, COWORK_SESSION_ID: 'test', COLORTERM: 'truecolor' } });
      return {
        cli_has_truecolor: /\x1b\[38;2/.test(cli.stdout),
        desktop_is_ascii: !/\x1b\[/.test(desktop.stdout),
        cowork_matches_cli: cli.stdout === cowork.stdout,
      };
    }

    function cross_check_all_tripwires() {
      const tripwires = [
        'audit-body-shape-coverage.cjs',
        'check-skill-vs-code-drift.cjs',
        'check-palette-consistency.cjs',
        'disposition-render-v2.cjs',
        'check-help-coverage.cjs',
      ];
      const results = {};
      for (const t of tripwires) {
        const r = spawnSync('node', [path.join(REPO, 'scripts', t)], { encoding: 'utf8' });
        results[t] = { exit_code: r.status, ok: r.status === 0 };
      }
      return results;
    }

    function four_questions_evidence_check(s1Result, s5Result) {
      // The four canonical questions
      const body = (s1Result && s1Result.envelope_text) || '';
      return {
        who_am_i_talking_to: /Larry/i.test(body),
        what_job_am_i_in: /JTBD|🎯/.test(body),
        how_much_runway: /📊|context/i.test(body),
        whats_next: /Next move|recommendation|F\.1|Action/i.test(body),
      };
    }

    function runAll() {
      const s1 = step1_cold_start();
      const s2 = step2_first_material_upload();
      const s3 = step3_mos_command();
      const s4 = step4_decision_gate();
      const s5 = step5_tension_resurface();
      const tri = tri_polar_checks();
      const trip = cross_check_all_tripwires();
      const four = four_questions_evidence_check(s1, s5);
      const pass = (
        s1.envelope_under_budget && s1.envelope_has_version && s1.banner_has_version &&
        s2.single_coherent_voice &&
        s3.all_groups_present &&
        s4.skill_drift_valid && s4.f_frame_has_filled_square && s4.f_frame_has_tri_context && s4.f_frame_has_free_text_last &&
        s5.envelope_under_budget &&
        tri.desktop_is_ascii && tri.cowork_matches_cli &&
        Object.values(trip).every(r => r.ok)
      );
      return {
        pass,
        step1_cold_start: s1,
        step2_first_material: s2,
        step3_mos_command: s3,
        step4_decision_gate: s4,
        step5_tension_resurface: s5,
        tri_polar: tri,
        all_tripwires: trip,
        four_questions: four,
      };
    }

    if (require.main === module) {
      const result = runAll();
      console.log(JSON.stringify(result, null, 2));
      process.exit(result.pass ? 0 : 1);
    }
    module.exports = { runAll, step1_cold_start, step2_first_material_upload, step3_mos_command, step4_decision_gate, step5_tension_resurface, tri_polar_checks, cross_check_all_tripwires, four_questions_evidence_check };
    ```

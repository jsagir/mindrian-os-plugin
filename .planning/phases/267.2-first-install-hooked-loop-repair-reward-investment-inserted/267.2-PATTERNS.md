```js
// lib/core/dual-path-detector.cjs:28-30, 32-77
const SECTION_HEADER = /^(Education|Experience|Summary|...|Pitch)\b/im;
const DOMAIN_MARKER  = /(IRB#?\s*\d|NCT\d{8}|NIH\b|...|portfolio)/i;
const STUCK_LANGUAGE = /(I'?m stuck|I can'?t|I don'?t know|I'?m trying to|help me think|I keep|...)/i;

function classify(text) {
  if (!text || typeof text !== 'string') {
    return { path: 'ambiguous', score: 0, features: {} };
  }
  const trimmed = text.trim();
  const words = trimmed.split(/\s+/).filter(Boolean);
  const wc = words.length;
  const newlines = (text.match(/\n/g) || []).length;
  const density = wc > 0 ? newlines / wc : 0;

  let score = 0;
  const features = {
    word_count: wc, newline_density: density,
    section_header: false, domain_marker: false, stuck_language: false,
  };

  if (wc < 80) score -= 2; else if (wc > 300) score += 2;          // F1
  if (density > 0.10) score += 2; else if (density < 0.05) score -= 1; // F2
  features.section_header = SECTION_HEADER.test(text); if (features.section_header) score += 3;
  features.domain_marker  = DOMAIN_MARKER.test(text);  if (features.domain_marker)  score += 2;
  features.stuck_language = STUCK_LANGUAGE.test(text); if (features.stuck_language) score -= 3; // NEGATIVE

  let path;
  if (score >= 3) path = 'upload'; else if (score <= -3) path = 'type'; else path = 'ambiguous';
  return { path, score, features };
}

module.exports = { classify, SECTION_HEADER, DOMAIN_MARKER, STUCK_LANGUAGE };
```

Its own header states the Part-8 contract to inherit verbatim (`:19-21`):
`Canon Part 8: pure string classification; output is enum + int + booleans (NO matched substrings
in features). Safe for Phase 121 trajectory telemetry which carries the features payload as scalars
only.`

**Inherit:** exported regex constants; a first-class `ambiguous` bucket; at least one
negative-weight feature that pulls AWAY from the aggressive outcome (here: away from firing
`/mos:ignite`); scalars-and-booleans-only `features`; the guard-clause return on non-string input.

**Do NOT inherit:** the single score axis. Two cut points give three regions, not four or five.

The multi-class alternative, and its live bug to avoid:

```js
// lib/core/user-archetype.cjs:154-172 (live)
let maxScore = 0;
let archetype = 'default';
for (const [name, score] of Object.entries(scores)) {
  if (score > maxScore) { maxScore = score; archetype = name; }   // <-- ties silently go to first-declared
}
const sortedScores = Object.values(scores).sort((a, b) => b - a);
const margin = sortedScores[0] - (sortedScores[1] || 0);

`score > maxScore` means a tie silently resolves to the first-declared key. That exact line is what
produces the C-3 wrong answer. F3 must return `ambiguous` on a tie or on a margin below threshold.

function writeUserMdAtomic(userMdPath, data) {
  let preservedBody = '\n# USER.md\n';
  try {
    const existing = fs.readFileSync(userMdPath, 'utf8');
    const fmRe = /^---\r?\n[\s\S]*?\r?\n---\r?\n?([\s\S]*)$/;
    const m = existing.match(fmRe);
    if (m && typeof m[1] === 'string') { preservedBody = m[1]; /* ...leading \n normalize... */ }
  } catch (_e) { /* no existing file */ }

  const newContent = buildFrontmatter(data) + '\n' + preservedBody.replace(/^\n/, '\n');
  const rnd = Math.random().toString(36).slice(2, 10);
  const tmpPath = userMdPath + '.tmp.' + process.pid + '.' + rnd + '.user';
  let fd;
  try { fd = fs.openSync(tmpPath, 'wx'); }
  catch (e) { if (e && e.code === 'EEXIST') { try { fs.unlinkSync(tmpPath); } catch (_) {} fd = fs.openSync(tmpPath, 'wx'); } else throw e; }
  try { fs.writeSync(fd, newContent); try { fs.fsyncSync(fd); } catch (_e) {} }
  finally { try { fs.closeSync(fd); } catch (_) {} }
  try { fs.renameSync(tmpPath, userMdPath); }
  catch (e) { try { fs.unlinkSync(tmpPath); } catch (_) {} throw e; }
}
```

Note `preservedBody` defaults to `'\n# USER.md\n'` - **this is the seam for C-3 fix option (a)**:
a prose body line the archetype regex can key on costs nothing structurally.
